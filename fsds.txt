-- === HELPERS ===
local wscon, slopecon
local function updws()
    if wscon then wscon:Disconnect() end
    if Options.wstoggle.Value or Options.jptoggle.Value then
        wscon = runs.RenderStepped:Connect(function()
            if hum then
                hum.WalkSpeed = Options.wstoggle.Value and Options.wsslider.Value or 16
                hum.JumpPower = Options.jptoggle.Value and Options.jpslider.Value or 50
            end
        end)
    end
end

local function updmsa()
    if slopecon then slopecon:Disconnect() end
    if Options.msatoggle.Value then
        slopecon = runs.RenderStepped:Connect(function()
            if hum then hum.MaxSlopeAngle = 90 end
        end)
    else
        if hum then hum.MaxSlopeAngle = 46 end
    end
end

Options.wstoggle:OnChanged(updws)
Options.jptoggle:OnChanged(updws)
Options.msatoggle:OnChanged(updmsa)

-- === UTILITY FUNCTIONS ===
local function getlayout(itemname)
    local inv = plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if not inv then return end
    for _, ch in ipairs(inv:GetChildren()) do
        if ch:IsA("ImageLabel") and ch.Name == itemname then return ch.LayoutOrder end
    end
end

local function swingtool(ids) if packets.SwingTool and packets.SwingTool.send then packets.SwingTool.send(ids) end end
local function pickup(id) if packets.Pickup and packets.Pickup.send then packets.Pickup.send(id) end end
local function drop(item) 
    local inv = plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if not inv then return end
    for _, ch in ipairs(inv:GetChildren()) do
        if ch:IsA("ImageLabel") and ch.Name == item then
            if packets.DropBagItem and packets.DropBagItem.send then
                packets.DropBagItem.send(ch.LayoutOrder)
            end
        end
    end
end

local selecteditems = {}

    selecteditems = {}
    for item, state in pairs(Value) do if state then table.insert(selecteditems, item) end end
end)

-- === SAFE GETTERS (avoid nil.Position) ===
local function getpbs(range)
    if not root then return {} end
    local boxes = {}
    for _, d in ipairs(workspace.Deployables:GetChildren()) do
        if d:IsA("Model") and d.Name == "Plant Box" then
            local eid = d:GetAttribute("EntityID")
            local ppart = d.PrimaryPart or d:FindFirstChildWhichIsA("BasePart")
            if eid and ppart then
                local dist = (ppart.Position - root.Position).Magnitude
                if dist <= range then table.insert(boxes, { eid = eid, deployable = d, dist = dist }) end
            end
        end
    end
    return boxes
end

local function getbushes(range, fruit)
    if not root then return {} end
    local bushes = {}
    for _, m in ipairs(workspace:GetChildren()) do
        if m:IsA("Model") and m.Name:find(fruit) then
            local ppart = m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")
            if ppart then
                local dist = (ppart.Position - root.Position).Magnitude
                if dist <= range then
                    local eid = m:GetAttribute("EntityID")
                    if eid then table.insert(bushes, { eid = eid, model = m, dist = dist }) end
                end
            end
        end
    end
    return bushes
end

-- 🎯 NON-STOP TWEEN ENGINE (Heartbeat-driven, no waits)
local tweening = nil
local function safeTween(targetCFrame)
    if not root then return end
    if tweening then tweening:Cancel() end
    local dist = (root.Position - targetCFrame.Position).Magnitude
    local speed = Options.tweenspeed.Value
    local duration = math.max(dist / speed, 0.05) -- min 0.05s to avoid teleport
    local info = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    tweening = tspmo:Create(root, info, { CFrame = targetCFrame })
    tweening:Play()
end

            local boxes = getpbs(range)
            for _, b in ipairs(boxes) do
                if not b.deployable:FindFirstChild("Seed") then
                    if packets.InteractStructure and packets.InteractStructure.send then
                        packets.InteractStructure.send({ entityID = b.eid, itemID = itemID })
                    end
                    break -- plant one per frame for max speed
                end
            end
        end)
    else
        if plantConn then plantConn:Disconnect() end
    end
end)

-- 🌿 Auto Harvest
task.spawn(function()
    while true do
        if not Options.harvesttoggle.Value then task.wait(0.1); continue end
        if not root then task.wait(0.1); continue end
        local range = tonumber(Options.harvestrange.Value) or 30
        local fruit = Options.fruitdropdown.Value
        local bushes = getbushes(range, fruit)
        table.sort(bushes, function(a,b) return a.dist < b.dist end)
        for i = 1, math.min(3, #bushes) do -- pick up to 3 per frame
            pickup(bushes[i].eid)
        end
        task.wait(0.05)
    end
end)

local plantBoxConn = nil
Options.tweentoplantbox:OnChanged(function(active)
    if active then
        if plantBoxConn then plantBoxConn:Disconnect() end
        plantBoxConn = runs.Heartbeat:Connect(function()
            if not root then return end
            local range = tonumber(Options.tweenrange.Value) or 250
            local boxes = getpbs(range)
            table.sort(boxes, function(a,b) return a.dist < b.dist end)
            for _, b in ipairs(boxes) do
                if not b.deployable:FindFirstChild("Seed") then
                    local target = b.deployable.PrimaryPart.CFrame + Vector3.new(0, 3, 0)
                    safeTween(target)
                    return -- move to *one* closest box per frame → non-stop flow
                end
            end
        end)
    else
        if plantBoxConn then plantBoxConn:Disconnect() end
        if tweening then tweening:Cancel() end
    end
end)

-- 🐾 NON-STOP TWEEN TO BUSH OR PLANT BOX
local bushConn = nil
Options.tweentobush:OnChanged(function(active)
    if active then
        if bushConn then bushConn:Disconnect() end
        bushConn = runs.Heartbeat:Connect(function()
            if not root then return end
            local range = tonumber(Options.tweenrange.Value) or 250
            local fruit = Options.fruitdropdown.Value
            local bushes = getbushes(range, fruit)
            table.sort(bushes, function(a,b) return a.dist < b.dist end)
            if #bushes > 0 then
                local target = bushes[1].model.PrimaryPart.CFrame + Vector3.new(0, 3, 0)
                safeTween(target)
                return
            end
            -- fallback: go to empty plant box
            local boxes = getpbs(range)
            table.sort(boxes, function(a,b) return a.dist < b.dist end)
            for _, b in ipairs(boxes) do
                if not b.deployable:FindFirstChild("Seed") then
                    local target = b.deployable.PrimaryPart.CFrame + Vector3.new(0, 3, 0)
                    safeTween(target)
                    return
                end
            end
        end)
    else
        if bushConn then bushConn:Disconnect() end
        if tweening then tweening:Cancel() end
    end
end)

-- 🌐 PLACE STRUCTURE (with 45° rotation via CFrame, and cursor fix)
local function placeSingleSnappedBox()
    if not root then warn("No root for placement"); return end
    local pos = root.Position
    local gridSize = 6.05
    local snapped = Vector3.new(
        math.round(pos.X / gridSize) * gridSize,
        pos.Y,
        math.round(pos.Z / gridSize) * gridSize
    )
    local ray = workspace:Raycast(snapped + Vector3.new(0, 100, 0), Vector3.new(0, -200, 0))
    local finalPos = ray and ray.Position or snapped

    -- ✅ Rotate CFrame directly (not yrot)
    local rot = CFrame.Angles(0, 0, 0)
    if Options.rotationdropdown.Value == "45 degrees" then
        rot = CFrame.Angles(0, math.rad(45), 0)
    end
    local cframe = CFrame.new(finalPos) * rot

    -- ✅ Add cursor = root.Position to prevent ByteNet crash
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({
            buildingName = "Plant Box",
            cframe = cframe,
            vec = finalPos,
            isMobile = false,
            cursor = root.Position -- CRITICAL
        })
        print("Placed at", finalPos, "rot:", Options.rotationdropdown.Value)
    end
end

local function placeGridOfBoxes(gridSize)
    if not root then return end
    local torso = root
    local start = torso.Position - Vector3.new(0, 3, 0)
    local spacing = 6.05
    local rot = Options.rotationdropdown.Value == "45 degrees" and CFrame.Angles(0, math.rad(45), 0) or CFrame.Angles(0, 0, 0)

    for x = 0, gridSize - 1 do
        for z = 0, gridSize - 1 do
            task.wait(0.15) -- small delay to avoid spamming server
            local gridPos = start + Vector3.new(x * spacing, 0, z * spacing)
            local snapped = Vector3.new(
                math.round(gridPos.X / spacing) * spacing,
                gridPos.Y,
                math.round(gridPos.Z / spacing) * spacing
            )
            local ray = workspace:Raycast(snapped + Vector3.new(0, 100, 0), Vector3.new(0, -200, 0))
            local finalPos = ray and ray.Position or snapped
            local cframe = CFrame.new(finalPos) * rot

            if packets.PlaceStructure and packets.PlaceStructure.send then
                packets.PlaceStructure.send({
                    buildingName = "Plant Box",
                    cframe = cframe,
                    vec = finalPos,
                    isMobile = false,
                    cursor = root.Position
                })
            end
        end
    end
end

placestructure = function(gridsize)
    if gridsize == 1 then
        placeSingleSnappedBox()
    else
        placeGridOfBoxes(gridsize)
    end
end
