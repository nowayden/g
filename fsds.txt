-- Helper function to format seconds into HH:MM:SS
local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- Robust inventory scanner that checks background data modules and UI layouts
local function getInventoryCount(itemName)
    if GameUtil and GameUtil.Data then
        local inv = GameUtil.Data.inventory or GameUtil.Data.Inventory or GameUtil.Data.bag or GameUtil.Data.Bag
        if type(inv) == "table" then
            if inv[itemName] then return tonumber(inv[itemName]) or 0 end
            for _, v in pairs(inv) do
                if type(v) == "table" and (v.name == itemName or v.Name == itemName) then
                    return tonumber(v.quantity or v.amount or v.Amount or v.count or v.Count or 1) or 0
                end
            end
        end
    end
    
    local list = plr.PlayerGui:FindFirstChild("MainGui") 
        and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") 
        and plr.PlayerGui:FindFirstChild("MainGui").RightPanel:FindFirstChild("Inventory") 
        and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if list then
        local total = 0
        for _, slot in ipairs(list:GetChildren()) do
            if slot:IsA("GuiObject") and (slot.Name == itemName or slot.Name:find(itemName)) then
                local quantityImg = slot:FindFirstChild("QuantityImage")
                local textObj = (quantityImg and quantityImg:FindFirstChild("QuantityText")) or slot:FindFirstChild("Amount") or slot:FindFirstChild("Count") or slot:FindFirstChildWhichIsA("TextLabel")
                if textObj then
                    total = total + (tonumber(textObj.Text:match("%d+")) or 1)
                else
                    total = total + 1
                end
            end
        end
        if total > 0 then return total end
    end
    return 0
end

-- Helper functions
local function swingtool(entityIds)
    if not (packets.SwingTool and packets.SwingTool.send) then return end
    local arr = {}
    for _, eid in ipairs(entityIds) do table.insert(arr, {entityID = eid, buffer = 0}) end
    pcall(function()
        packets.SwingTool.send({entityIDs = arr, cframe = root and root.CFrame or CFrame.new(), timestamp = clock.getServerTime(true)})
    end)
end

-- Resource Aura
local resourceActive, resourceConn, lastResourceTime = false, nil, 0
local function startResource()
    if resourceActive then return end
    resourceActive = true
    resourceConn = runs.Heartbeat:Connect(function()
        if not Options.resourceauratoggle.Value then resourceConn:Disconnect(); resourceConn = nil; resourceActive = false; return end
        if not root or not GameUtil.Data.equipped then return end
        local ct = tick()
        if ct - lastResourceTime < (tonumber(Options.resourcecooldownslider.Value) or 0.1) then return end
        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1
        local targets, all = {}, {}
        
        if workspace:FindFirstChild("Resources") then
            for _, r in pairs(workspace.Resources:GetDescendants()) do 
                if r:IsA("Model") then 
                    table.insert(all, r) 
                end 
            end
        end
 
        local resourceKeywords = {"Node", "Stone", "Lode", "Rock", "Coal", "Guardian"}
        for _, r in pairs(workspace:GetChildren()) do 
            if r:IsA("Model") then
                local matchFound = false
                for _, kw in ipairs(resourceKeywords) do
                    if string.find(r.Name, kw) then
                        matchFound = true
                        break
                    end
                end
                 if matchFound then
                    table.insert(all, r)
                end
            end 
        end
        
        for _, res in pairs(all) do
            if res:IsA("Model") and res:GetAttribute("EntityID") then
                local pp = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                if pp and (pp.Position - root.Position).Magnitude <= range then
                    table.insert(targets, {eid = res:GetAttribute("EntityID"), dist = (pp.Position - root.Position).Magnitude})
                end
            end
        end
        if #targets > 0 then
            table.sort(targets, function(a,b) return a.dist < b.dist end)
            local sel = {}
            for i = 1, math.min(tc, #targets) do table.insert(sel, targets[i].eid) end
            swingtool(sel)
            lastResourceTime = ct
        end
    end)
end
Options.resourceauratoggle:OnChanged(function(s) if s then startResource() else resourceActive = false; if resourceConn then resourceConn:Disconnect(); resourceConn = nil end end end)

local RS = runs
local isTraderActive = false

local deployableAuraActive = false
local deployableAuraConnection = nil
local lastDeployableSwingTime = 0

function StartDeployableAura(range)
    if deployableAuraActive then return end
    deployableAuraActive = true
    range = range or 25

    deployableAuraConnection = RS.Heartbeat:Connect(function()
        if not deployableAuraActive or isTraderActive then return end

        local currentTime = tick()
        if currentTime - lastDeployableSwingTime < 0.1 then return end

        local char = plr.Character
        local rootPart = char and char:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end

        local targetList = {}
        local folders = {"Deployables", "Rubble", "dropChests"}
      
        for _, folderName in ipairs(folders) do
            local folder = workspace:FindFirstChild(folderName)
             if folder then
                for _, child in pairs(folder:GetChildren()) do
                    local eid = child:GetAttribute("EntityID")
                    if eid then
                        local pp = child.PrimaryPart or child:FindFirstChildWhichIsA("BasePart")
                        if pp and (pp.Position - rootPart.Position).Magnitude <= range then
                            table.insert(targetList, {entityID = eid, buffer = 0})
                        end
                    end
                end
            end
        end

        if #targetList > 0 then
            pcall(function()
                 packets.SwingTool.send({
                    entityIDs = targetList,
                    cframe = rootPart.CFrame,
                    timestamp = clock.getServerTime(true)
                })
            end)
            lastDeployableSwingTime = currentTime
        end
    end)
end

function StopDeployableAura()
    deployableAuraActive = false
    if deployableAuraConnection then deployableAuraConnection:Disconnect() end
end

local function getpbs(range)
    local plantboxes = {}
    if not root then return plantboxes end
    local deployablesFolder = workspace:FindFirstChild("Deployables")
    if not deployablesFolder then return plantboxes end
    local charPos = root.Position
    local children = deployablesFolder:GetChildren()
     for i = 1, #children do
        local deployable = children[i]
        if deployable.Name == "Plant Box" and deployable:IsA("Model") then
            local ppart = deployable.PrimaryPart or deployable:FindFirstChildWhichIsA("BasePart")
            if ppart then
                local dist = (ppart.Position - charPos).Magnitude
                 if dist <= range then
                    local entityid = deployable:GetAttribute("EntityID")
                    if entityid then table.insert(plantboxes, { entityid = entityid, deployable = deployable, dist = dist }) end
                end
            end
        end
    end
    return plantboxes
end

local function getbushes(range, fruitname)

    for i = 1, #children do

                end
            end
        end
    end
    return bushes
end

local lastPos = Vector3.new()
local stuckTimer = 0
local lastCheck = 0

local function walkToTarget(targetPosition)
    if hum and root and targetPosition then
        hum:MoveTo(targetPosition)
         local now = tick()
         local dt = now - lastCheck
        if lastCheck == 0 or dt > 0.5 then
            lastCheck = now
            lastPos = root.Position
            stuckTimer = 0
            return
        end
        lastCheck = now
         if (root.Position - targetPosition).Magnitude > 3 then
            if (root.Position - lastPos).Magnitude < 1 * dt then
                stuckTimer = stuckTimer + dt
                if stuckTimer >= 3 then
                     hum.Jump = true
                      stuckTimer = 0
                end
            else
                stuckTimer = 0
            end
        else
            stuckTimer = 0
       end
        lastPos = root.Position
    end
end

local function pathToPlantbox(range)
    local plantboxes = getpbs(range)
    table.sort(plantboxes, function(a, b) return a.dist < b.dist end)
    for _, box in ipairs(plantboxes) do
        if not box.deployable:FindFirstChild("Seed") then
            if box.deployable.PrimaryPart then
                walkToTarget(box.deployable.PrimaryPart.Position)
                break
            end
        end
    end
end

local function pathToBushesOrBoxes(range, fruitname)
    local bushes = getbushes(range, fruitname)
    table.sort(bushes, function(a, b) return a.dist < b.dist end)
    if #bushes > 0 then
        for _, bush in ipairs(bushes) do
            if bush.model.PrimaryPart then
                walkToTarget(bush.model.PrimaryPart.Position)
                break
            end
        end
    else
        local plantboxes = getpbs(range)
        table.sort(plantboxes, function(a, b) return a.dist < b.dist end)
        for _, box in ipairs(plantboxes) do
            if not box.deployable:FindFirstChild("Seed") then
                if box.deployable.PrimaryPart then
                    walkToTarget(box.deployable.PrimaryPart.Position)
                    break
                end
            end
        end
    end
end

-- Auto Plant: same logic, but with a fixed interval to avoid hammering FPS
local lastPlantCacheUpdate = 0
local plantCache = {}
local PLANT_CHECK_INTERVAL = 0.05  -- tweak this if you want more/less aggressiveness

task.spawn(function()
    local lastPlantCheck = 0
    while true do
        if not Options.planttoggle.Value then
            task.wait(0.1)
            continue
        end

        local now = tick()
        if now - lastPlantCheck < PLANT_CHECK_INTERVAL then
            task.wait(PLANT_CHECK_INTERVAL)
            continue
        end
        lastPlantCheck = now

        local fruitName = Options.fruitdropdown.Value
        local itemID = fruittoitemid[fruitName]
        if not itemID then
            task.wait(0.1)
            continue
        end

        local range = tonumber(Options.plantrange.Value) or 30
        local deployablesFolder = workspace:FindFirstChild("Deployables")
        if deployablesFolder then
            local currentTime = tick()
            if currentTime - lastPlantCacheUpdate > 2 then
                plantCache = {}
                local children = deployablesFolder:GetChildren()
                for i = 1, #children do
                    local v = children[i]
                    if v.Name == "Plant Box" and v:IsA("Model") then
                        local ppart = v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")
                        local entityid = v:GetAttribute("EntityID")
                        if ppart and entityid then
                            table.insert(plantCache, {box = v, part = ppart, id = entityid})
                        end
                    end
                end
                lastPlantCacheUpdate = currentTime
            end

            if root then
                local myPos = root.Position
                for i = 1, #plantCache do
                    local data = plantCache[i]
                    if data.box.Parent == deployablesFolder then
                        if (data.part.Position - myPos).Magnitude <= range then
                            if not data.box:FindFirstChild("Seed") then
                                plant(data.id, itemID)
                            else
                                plantedboxes[data.id] = true
                            end
                        end
                    end
                end
            end
        end

        task.wait(PLANT_CHECK_INTERVAL)
    end
end)

-- REWRITTEN HIGH-ACCURACY PARALLEL TRACKING LOOP
task.spawn(function()
    local lastCount = nil
    local trackedBushes = {}
    while true do
        local targetFruit = Options.fruitdropdown and Options.fruitdropdown.Value or "Bloodfruit"
        local currentCount = getInventoryCount(targetFruit) or 0
        local gained = 0
        
        if lastCount == nil then
            lastCount = currentCount
            print("[Tracker] Context Engine Init baseline total for " .. targetFruit .. ": " .. tostring(lastCount))
        else
            if currentCount > lastCount then
                gained = currentCount - lastCount
                print("[Tracker Inventory Match] Found Increment: +" .. tostring(gained) .. " |\nItem: " .. targetFruit)
                lastCount = currentCount
            elseif currentCount < lastCount then
                print("[Tracker Inventory Recalibrate] Drop/Spent detected. Adjusting inventory baseline down to: " .. tostring(currentCount))
                lastCount = currentCount
            end
        end
        
        if Options.harvesttoggle.Value and targetFruit == "Bloodfruit" and root then
            local harvestrange = tonumber(Options.harvestrange.Value) or 30
            local bushes = getbushes(harvestrange, "Bloodfruit")
            for id, data in pairs(trackedBushes) do
                if not workspace:FindFirstChild(data.name) and (data.pos - root.Position).Magnitude <= harvestrange then
                    if gained == 0 then
                        gained = 3
                        print("[Tracker Fallback Match] Bush deletion registered within radius! Force-adding +3 fruits.")
                    end
                    trackedBushes[id] = nil
                end
            end
            for _, b in ipairs(bushes) do
                if b.model.PrimaryPart then
                    trackedBushes[b.entityid] = { name = b.model.Name, pos = b.model.PrimaryPart.Position }
                end
            end
        end
        
        if gained > 0 then
            farmStats.timerStarted = true
            if targetFruit == "Bloodfruit" then
                farmStats.bloodfruits = farmStats.bloodfruits + gained
            end
            farmStats.totalAllTime = farmStats.totalAllTime + gained
        end
        
        local elapsed = farmStats.timerStarted and (tick() - farmStats.startTime) or 0
        local totalSessionElapsed = farmStats.timerStarted and (tick() - farmStats.sessionStartTime) or 0
        if elapsed < 1 then elapsed = 1 end
        local bloodfruitsPerMin = farmStats.timerStarted and math.round((farmStats.bloodfruits / elapsed) * 60) or 0
        local bloodfruitsPerHour = farmStats.timerStarted and math.round((farmStats.bloodfruits / elapsed) * 3600) or 0
        
        pcall(function()
            BloodfruitParagraph:SetContent(string.format("%d (Per Min: %d |\nPer Hour: %d)", farmStats.bloodfruits, bloodfruitsPerMin, bloodfruitsPerHour))
            if farmStats.timerStarted then

            else
                TotalAllTimeParagraph:SetContent(string.format("%d (Time: 00:00:00 (Waiting for harvest...))", farmStats.totalAllTime))
            end
        end)
        task.wait(0.3)
    end
end)

-- Plant box placement (Tight Edge-to-Edge Snapping)
function placeSingleSnappedBox()
    local char = plr.Character
    if not (char and char:FindFirstChild("HumanoidRootPart")) then return end
    local pos = char.HumanoidRootPart.Position
    local cam = workspace.CurrentCamera
    local cursorPos = cam and cam.CFrame.Position or pos
    local gs = 5.05
    local sp = Vector3.new(math.round(pos.X/gs)*gs, pos.Y, math.round(pos.Z/gs)*gs)
    local rp = RaycastParams.new()
    rp.FilterType, rp.FilterDescendantsInstances, rp.IgnoreWater = Enum.RaycastFilterType.Blacklist, {char}, true
    local rr = workspace:Raycast(Vector3.new(sp.X, sp.Y+100, sp.Z), Vector3.new(0,-200,0), rp)
    local fp = rr and rr.Position or sp
    local rot = CFrame.Angles(0, 0, 0)
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({buildingName = "Plant Box", cframe = CFrame.new(fp)*rot, vec = fp, isMobile = false, cursor = cursorPos})
    end
end

local placementConn
Options.placeplantboxestoggle:OnChanged(function(v)

        placementConn = runs.Heartbeat:Connect(function()
            placeSingleSnappedBox()
            task.wait(0.1)
        end)
    else
        if placementConn then placementConn:Disconnect(); placementConn = nil end
    end
end)

local function findAllCauldrons()
    local dep = workspace:FindFirstChild("Deployables")
    if not dep or not (plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")) then return {} end
    local rp = plr.Character.HumanoidRootPart.Position
    local cauldrons = {}
    for _, obj in ipairs(dep:GetChildren()) do
        if obj.Name == "Cauldron" and obj:FindFirstChild("Hitbox") and (obj.Hitbox.Position - rp).Magnitude < 30 then
            table.insert(cauldrons, {cauldron = obj, hitbox = obj.Hitbox})
        end
    end
    return cauldrons
end

local function teleportToHitbox(hitbox)
    if not hitbox or not workspace:FindFirstChild("Items") then return end
    local tp = hitbox.Position + Vector3.new(math.random(-2,2)*0.3, 0.8, math.random(-2,2)*0.3)
    for _, item in ipairs(workspace.Items:GetChildren()) do
        local n = item.Name or ""
        if n:find("Ice Cube") or n:find("Bloodfruit") or n:find("Strawberry") or n:find("Prickly Pear") or n:find("Crystal") or n:find("Cloudberry") or n:find("Adurite") or n:find("Fire Hide") or n:find("Berry") or n:find("Iron") or n:find("Lemon") or n:find("Magnetite") then
            pcall(function()
                if item:IsA("BasePart") then
                    item.Position = tp
                elseif item:IsA("Model") and item.PrimaryPart then
                    item:SetPrimaryPartCFrame(CFrame.new(tp))
                end
            end)
        end
    end
end

local function brewPotions()
    if isBrewing then return end
    isBrewing = true
    local cauldrons = findAllCauldrons()
    if #cauldrons == 0 then isBrewing = false; return end
    local recipe = POTIONS[selectedPotion]
    if not recipe then isBrewing = false; return end
    
    for _, ing in ipairs(recipe) do
        local slot = getInventorySlot(ing.name)
        if slot then
            for _, c in ipairs(cauldrons) do
                dropFromSlot(slot, ing.amount)
                task.wait(0.5)
                teleportToHitbox(c.hitbox)
                task.wait(0.4)
            end
        end
        task.wait(0.8)
    end
    task.wait(2)
    for _, c in ipairs(cauldrons) do
        local eid = c.cauldron:GetAttribute("EntityID")
        if eid then
            pcall(function()
                if packets.InteractStructure and packets.InteractStructure.send then
                    packets.InteractStructure.send({ entityID = eid, itemID = 0 })
                end
            end)
        end
    end
    isBrewing = false
end

local function getPlayerLevel()
    local success, GameUtilLocal = pcall(function() return require(rs.Modules.GameUtil) end)
    if success and GameUtilLocal.Data and GameUtilLocal.Data.level then
        return GameUtilLocal.Data.level
    end
    return plr:GetAttribute("Level") or 0
end

local function doRebirth()
    local success, packetsLocal = pcall(function() return require(rs.Modules.Packets) end)
    if success and packetsLocal.Rebirth then
        pcall(function() packetsLocal.Rebirth.send() end)
    end
    task.wait(2.5)
    pressBedButton()
end

local function AutoRebirthLoop()
    while autoRebirthRunning do
        task.wait(0.5)
        local spawnGui = plr.PlayerGui:FindFirstChild("SpawnGui")
        if spawnGui and spawnGui.Enabled then
            pressBedButton()
        end
        local currentLevel = getPlayerLevel()
        if currentLevel >= 100 then
            doRebirth()
            task.wait(6)
        end
    end
end

local function StartAutoRebirth()
    if autoRebirthRunning then return end
    autoRebirthRunning = true
    task.spawn(AutoRebirthLoop)
end

-- Prioritized Targets TP Execution System (Soul Minion Tracker Integration)
local function getPrioritizedTarget()
    local critters = workspace:FindFirstChild("Critters")
    if not critters then return nil end
    
    local traderInstance, queenInstance, minionInstance, antInstance
    local traderMinDist, queenMinDist, minionMinDist, antMinDist = math.huge, math.huge, math.huge, math.huge
    
    for _, c in ipairs(critters:GetChildren()) do
        if c:IsA("Model") and c:GetAttribute("EntityID") then
            local pp = c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart")
            if pp and root then
                local dist = (pp.Position - root.Position).Magnitude
                
                if (c.Name:find("Trader") or c.Name == "Mysterious Trader") then
                    if dist < traderMinDist then
                        traderMinDist = dist
                        traderInstance = c
                    end
                elseif c.Name == "Voodoo Spider Queen" and not traderInstance then
                    if dist < queenMinDist then
                        queenMinDist = dist
                        queenInstance = c
                    end
                elseif c.Name == "Soul Minion" and not traderInstance and not queenInstance then
                    if dist < minionMinDist then
                        minionMinDist = dist
                        minionInstance = c
                    end
                elseif c.Name == "Void Ant" and not traderInstance and not queenInstance and not minionInstance then
                    if dist < antMinDist then
                        antMinDist = dist
                        antInstance = c
                    end
                end
            end
        end
    end
    
    if traderInstance then return traderInstance end
    if queenInstance then return queenInstance end
    if minionInstance then return minionInstance end
    if antInstance then return antInstance end
    return nil
end

    local targetExpiryDelay = 0
    
    while true do
        if Options.tpsoulminionstoggle and Options.tpsoulminionstoggle.Value and root then
            local rightNow = tick()
            if rightNow >= targetExpiryDelay then
                local target = getPrioritizedTarget()
                if target and target.PrimaryPart then
                    root.CFrame = target.PrimaryPart.CFrame * CFrame.new(0, 0, 3)
                end
            end
        end
        task.wait(0.1)
    end
end)
