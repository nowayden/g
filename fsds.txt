-- CORRECT SWING FUNCTION using the actual swing packet
-- From the toolbar script, we can see they use packets like:
-- EquipTool.send, ToolSwap.send, etc.
-- We need to find the actual swing/attack packet

-- Let's check what swing-related packets exist
local swingPacket = nil
local possibleSwingPackets = {"Swing", "Attack", "UseTool", "ToolSwing", "Hit", "Damage", "Mine", "Chop"}

for _, name in ipairs(possibleSwingPackets) do
    if packets[name] and packets[name].send then
        swingPacket = packets[name]
        print("Found swing packet: " .. name)
        break
    end
end

-- If we still don't have one, try to find any packet that might be used for swinging
if not swingPacket then
    print("Looking for any packet with send method...")
    for name, packet in pairs(packets) do
        if type(packet) == "table" and packet.send and not string.find(name, "Pickup") and not string.find(name, "Drop") then
            print("Found potential packet: " .. name)
            if not swingPacket then
                swingPacket = packet
            end
        end
    end
end

-- Create the swing function
local function swingTool(targetEntityIds)
    if swingPacket and swingPacket.send then
        -- Get cursor position
        local cursorPos = Vector3.new(0, 0, 0)
        local mouse = plr:GetMouse()
        if mouse and mouse.Hit then
            cursorPos = mouse.Hit.Position
        else
            local camera = workspace.CurrentCamera
            if camera then
                cursorPos = camera.CFrame.Position
            end
        end
        
        -- Try to send with different formats
        pcall(function()
            -- Format 1: just the entity IDs
            swingPacket.send(targetEntityIds)
        end)
        
        pcall(function()
            -- Format 2: with cursor
            swingPacket.send({
                targetEntityIds = targetEntityIds,
                cursor = cursorPos
            })
        end)
        
        pcall(function()
            -- Format 3: as an array
            swingPacket.send({ targetEntityIds = targetEntityIds })
        end)
    else
        -- If no packet found, try to use the old SwingTool packet as fallback
        if packets.SwingTool and packets.SwingTool.send then
            pcall(function()
                packets.SwingTool.send({
                    targetEntityIds = targetEntityIds
                })
            end)
        end
    end
end

-- Resource Aura
task.spawn(function()
    while true do
        if Options.resourceauratoggle.Value and root then
            local range = Options.resourceaurarange.Value
            local targetCount = tonumber(Options.resourcetargetdropdown.Value)
            local cooldown = Options.resourcecooldownslider.Value
            local targets = {}
            
            local resourcesFolder = workspace:FindFirstChild("Resources")
            if resourcesFolder then
                for _, res in pairs(resourcesFolder:GetChildren()) do
                    if res:IsA("Model") then
                        local eid = res:GetAttribute("EntityID")
                        if eid then
                            local ppart = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                            if ppart and root then
                                local dist = (ppart.Position - root.Position).Magnitude
                                if dist <= range then
                                    table.insert(targets, { eid = eid, dist = dist })
                                end
                            end
                        end
                    end
                end
            end
            
            if #targets > 0 then
                table.sort(targets, function(a, b) return a.dist < b.dist end)
                local selectedTargets = {}
                for i = 1, math.min(targetCount, #targets) do
                    table.insert(selectedTargets, targets[i].eid)
                end
                swingTool(selectedTargets)
            end
            task.wait(cooldown)
        else
            task.wait(0.1)
        end
    end
end)

local fruittoitemid = {
    Bloodfruit = 94, Bluefruit = 377, Lemon = 99, Coconut = 1, Jelly = 604,
    Banana = 606, Orange = 602, Oddberry = 32, Berry = 35, Strangefruit = 302,
    Strawberry = 282, Sunfruit = 128, Pumpkin = 80, ["Prickly Pear"] = 378,

local function plant(entityid, itemID)
    if packets.InteractStructure and packets.InteractStructure.send then
        pcall(function() packets.InteractStructure.send({ entityID = entityid, itemID = itemID }) end)
    end
end

local function getpbs(range)
    local plantboxes = {}
    if not root then return plantboxes end
    local deployablesFolder = workspace:FindFirstChild("Deployables")
    if deployablesFolder then
        for _, deployable in ipairs(deployablesFolder:GetChildren()) do
            if deployable:IsA("Model") and deployable.Name == "Plant Box" then
                local entityid = deployable:GetAttribute("EntityID")
                local ppart = deployable.PrimaryPart or deployable:FindFirstChildWhichIsA("BasePart")
                if entityid and ppart and root then 
                    local dist = (ppart.Position - root.Position).Magnitude
                    if dist <= range then
                        table.insert(plantboxes, { entityid = entityid, deployable = deployable, dist = dist })
                    end
                end
            end
        end
    end
    return plantboxes
end

local function getbushes(range, fruitname)
    local bushes = {}
    if not root then return bushes end
    for _, model in ipairs(workspace:GetChildren()) do
        if model:IsA("Model") and model.Name and model.Name:find(fruitname) then
            local ppart = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            if ppart and root then 
                local dist = (ppart.Position - root.Position).Magnitude
                if dist <= range then
                    local entityid = model:GetAttribute("EntityID")
                    if entityid then
                        table.insert(bushes, { entityid = entityid, model = model, dist = dist })
                    end
                end
            end
        end
    end
    return bushes
end

-- Tweening
local tweening = nil
local function tween(target)
    if not root then return end
    if tweening then tweening:Cancel() end
    local distance = (root.Position - target.Position).Magnitude
    local duration = math.max(distance / Options.tweenspeed.Value, 0.1)
    tweening = tspmo:Create(root, TweenInfo.new(duration, Enum.EasingStyle.Linear), { CFrame = target })
    tweening:Play()
    tweening.Completed:Connect(function() tweening = nil end)
end

local function tweenplantbox(range)
    if not root then return end
    local plantboxes = getpbs(range)
    local closest = nil
    for _, box in ipairs(plantboxes) do
        if not box.deployable:FindFirstChild("Seed") then
            if not closest or box.dist < closest.dist then
                closest = box
            end
        end
    end
    if closest then
        tween(closest.deployable.PrimaryPart.CFrame + Vector3.new(0, 3, 0))
    end
end

local function tweenpbs(range)
    if not root then return end
    local fruit = Options.fruitdropdown.Value
    local bushes = getbushes(range, fruit)
    if #bushes > 0 then
        table.sort(bushes, function(a,b) return a.dist < b.dist end)
        tween(bushes[1].model.PrimaryPart.CFrame + Vector3.new(0, 3, 0))
        if bushes[1].dist <= range then pickup(bushes[1].entityid) end
    else
        local plantboxes = getpbs(range)
        local closest = nil
        for _, box in ipairs(plantboxes) do
            if not box.deployable:FindFirstChild("Seed") then
                if not closest or box.dist < closest.dist then
                    closest = box
                end
            end
        end
        if closest then
            tween(closest.deployable.PrimaryPart.CFrame + Vector3.new(0, 3, 0))
            if closest.dist <= range then
                plant(closest.entityid, fruittoitemid[fruit] or 94)
            end
        end
    end
end

-- Plant Box Placement
local placementConnection
local function placeSingleSnappedBox()
    local character = plr.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    local rootPart = character.HumanoidRootPart
    local currentPosition = rootPart.Position
    local camera = workspace.CurrentCamera
    local cursorPos = camera and camera.CFrame.Position or currentPosition
    
    local gridSize = 6.05
    local snappedX = math.round(currentPosition.X / gridSize) * gridSize
    local snappedZ = math.round(currentPosition.Z / gridSize) * gridSize
    local snappedPosition = Vector3.new(snappedX, currentPosition.Y, snappedZ)
    
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    rayParams.FilterDescendantsInstances = {character}
    local rayResult = workspace:Raycast(snappedPosition + Vector3.new(0,100,0), Vector3.new(0,-200,0), rayParams)
    local finalPosition = rayResult and rayResult.Position or snappedPosition
    
    local rotationCFrame = (Options.rotationdropdown.Value == "45 degrees") and CFrame.Angles(0, math.rad(45), 0) or CFrame.Angles(0,0,0)
    local finalCFrame = CFrame.new(finalPosition) * rotationCFrame
    
    if packets.PlaceStructure and packets.PlaceStructure.send then
        pcall(function()
            packets.PlaceStructure.send({
                buildingName = "Plant Box",
                cframe = finalCFrame,
                vec = finalPosition,
                isMobile = false,
                cursor = cursorPos
            })
        end)
    end
end

local function placeGridOfBoxes(gridSize)
    local character = plr.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    local torso = character.HumanoidRootPart
    local cursorPos = workspace.CurrentCamera and workspace.CurrentCamera.CFrame.Position or torso.Position
    local rotationCFrame = (Options.rotationdropdown.Value == "45 degrees") and CFrame.Angles(0, math.rad(45), 0) or CFrame.Angles(0,0,0)
    local startpos = torso.Position - Vector3.new(0, 3, 0)
    local spacing = 6.05
    
    for x = 0, gridSize - 1 do
        for z = 0, gridSize - 1 do
            task.wait(0.3)
            local gridPos = Vector3.new(startpos.X + x*spacing, startpos.Y, startpos.Z + z*spacing)
            local snappedX = math.round(gridPos.X / spacing) * spacing
            local snappedZ = math.round(gridPos.Z / spacing) * spacing
            local snappedPos = Vector3.new(snappedX, gridPos.Y, snappedZ)
            
            local rayResult = workspace:Raycast(snappedPos + Vector3.new(0,100,0), Vector3.new(0,-200,0))
            local finalPos = rayResult and rayResult.Position or snappedPos
            local finalCFrame = CFrame.new(finalPos) * rotationCFrame
            
            if packets.PlaceStructure and packets.PlaceStructure.send then
                pcall(function()
                    packets.PlaceStructure.send({
                        buildingName = "Plant Box",
                        cframe = finalCFrame,
                        vec = finalPos,
                        isMobile = false,
                        cursor = cursorPos
                    })
                end)
            end
        end
    end
end

placestructure = function(gridsize)
    if gridsize == 1 then placeSingleSnappedBox() else placeGridOfBoxes(gridsize) end
end

Options.placeplantboxestoggle:OnChanged(function(v)
    if v then
        if placementConnection then placementConnection:Disconnect() end
        placementConnection = runs.Heartbeat:Connect(function()
            placeSingleSnappedBox()
            task.wait(0.1)

    elseif placementConnection then
        placementConnection:Disconnect()
    end
end)
