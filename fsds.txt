local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- Caching inventory lookups to prevent UI tree traversal lag
local cachedInventoryCount = 0
task.spawn(function()
    while true do
        task.wait(0.3) -- Pull inventory data 3 times a second instead of every frame
        local targetFruit = Options.fruitdropdown and Options.fruitdropdown.Value
        if not targetFruit then continue end
        
        local count = 0
        if GameUtil and GameUtil.Data then
            local inv = GameUtil.Data.inventory or GameUtil.Data.Inventory or GameUtil.Data.bag or GameUtil.Data.Bag
            if type(inv) == "table" then
                if inv[targetFruit] then 
                    count = tonumber(inv[targetFruit]) or 0 
                else
                    for k, v in pairs(inv) do
                        if k == targetFruit then count = tonumber(v) or 0; break end
                        if type(v) == "table" and (v.name == targetFruit or v.Name == targetFruit) then
                            count = tonumber(v.amount or v.Amount or v.count or v.Count or 1) or 0
                            break
                        end
                    end
                end
            end
        end
        cachedInventoryCount = count
    end
end)

-- Fully Optimized WalkSpeed/JumpPower Handler (Removes RenderStepped strain)
local function updws()
    if wscon then wscon:Disconnect(); wscon = nil end
    if Options.wstoggle.Value or Options.jptoggle.Value then
        wscon = runs.Heartbeat:Connect(function() -- Shifted from RenderStepped to Heartbeat
            if hum then
                local tWS = Options.wstoggle.Value and Options.wsslider.Value or 16
                local tJP = Options.jptoggle.Value and Options.jpslider.Value or 50
                if hum.WalkSpeed ~= tWS then hum.WalkSpeed = tWS end
                if hum.JumpPower ~= tJP then hum.JumpPower = tJP end
            end
        end)
    end
end

-- Network helpers
local function swingtool(entityIds)
    if not (packets.SwingTool and packets.SwingTool.send) then return end
    local arr = table.create(#entityIds)
    for i = 1, #entityIds do arr[i] = {entityID = entityIds[i], buffer = 0} end
    pcall(function()
        packets.SwingTool.send({entityIDs = arr, cframe = root and root.CFrame or CFrame.new(), timestamp = clock.getServerTime(true)})
    end)
end

-- Optimized Resource Aura
local resourceActive, resourceConn, lastResourceTime = false, nil, 0
local function startResource()
    if resourceActive then return end
    resourceActive = true
    resourceConn = runs.Heartbeat:Connect(function()
        if not Options.resourceauratoggle.Value then resourceConn:Disconnect(); resourceConn = nil; resourceActive = false; return end
        if not root or not GameUtil.Data.equipped then return end
        local ct = tick()
        local cd = tonumber(Options.resourcecooldownslider.Value) or 0.1
        if ct - lastResourceTime < cd then return end
        
        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1
        local targets = {}
        local rPos = root.Position

        local function checkFolder(folder)
            if not folder then return end
            local children = folder:GetChildren()
            for i = 1, #children do
                local res = children[i]
                local eid = res:GetAttribute("EntityID")
                if eid then
                    local pp = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                    if pp then
                        local dist = (pp.Position - rPos).Magnitude
                        if dist <= range then
                            table.insert(targets, {eid = eid, dist = dist})
                        end
                    end
                end
            end
        end

        checkFolder(workspace.Resources)
        checkFolder(workspace) -- Captures manual loose nodes like Gold Node models

        if #targets > 0 then
            table.sort(targets, function(a,b) return a.dist < b.dist end)
            local sel = table.create(math.min(tc, #targets))
            for i = 1, math.min(tc, #targets) do sel[i] = targets[i].eid end
            swingtool(sel)
            lastResourceTime = ct
        end
    end)
end
Options.resourceauratoggle:OnChanged(function(s) if s then startResource() else resourceActive = false; if resourceConn then resourceConn:Disconnect(); resourceConn = nil end end end)

            if deployablesFolder then
                local elements = deployablesFolder:GetChildren()
                for i = 1, #elements do
                    local chest = elements[i]
                    if chest.Name == "Chest" and chest:FindFirstChild("Contents") and chest.PrimaryPart then
                        if (chest.PrimaryPart.Position - rPos).Magnitude <= range then
                            local contents = chest.Contents:GetChildren()

                                if item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then
                                    pickup(item:GetAttribute("EntityID"))
                                end
                            end
                        end
                    end
                end
            end
        end
    end
end)

-- Shared Caching Structures for Farming Paths to prevent performance leaks
local plantBoxesCache = {}
local fruitBushesCache = {}
local fruittoitemid = {Bloodfruit = 94, Bluefruit = 377, Lemon = 99, Coconut = 1, Jelly = 604, Banana = 606, Orange = 602, Oddberry = 32, Berry = 35, Strangefruit = 302, Strawberry = 282, Sunfruit = 128, Pumpkin = 80, ["Prickly Pear"] = 378, Apple = 243, Barley = 247, Cloudberry = 101, Carrot = 147, Blossom = 958}

task.spawn(function()
    while true do
        task.wait(0.3) -- Build path targets cache every 300ms 
        if not root then continue end
        local rPos = root.Position
        local range = tonumber(Options.tweenrange.Value) or 250
        local targetFruit = Options.fruitdropdown.Value
        
        -- Plantbox Scanner Cache
        table.clear(plantBoxesCache)
        local deployables = workspace:FindFirstChild("Deployables")
        if deployables then
            local children = deployables:GetChildren()
            for i = 1, #children do
                local v = children[i]
                if v.Name == "Plant Box" and v:IsA("Model") then
                    local ppart = v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")
                    local eid = v:GetAttribute("EntityID")
                    if ppart and eid then
                        local dist = (ppart.Position - rPos).Magnitude
                        if dist <= range then
                            table.insert(plantBoxesCache, {entityid = eid, deployable = v, dist = dist, pos = ppart.Position})
                        end
                    end
                end
            end
            table.sort(plantBoxesCache, function(a,b) return a.dist < b.dist end)
        end

        -- Fruit Bush Scanner Cache
        table.clear(fruitBushesCache)
        local scanTargets = {}
        local wChildren = workspace:GetChildren()
        for i = 1, #wChildren do
            local model = wChildren[i]
            if model:IsA("Model") and model.Name:find(targetFruit) then table.insert(scanTargets, model) end
        end
        if deployables then
            local children = deployables:GetChildren()
            for i = 1, #children do
                local box = children[i]
                if box.Name == "Plant Box" then
                    local bChildren = box:GetChildren()
                    for j = 1, #bChildren do
                        local child = bChildren[j]
                        if child:IsA("Model") and child.Name:find(targetFruit) then table.insert(scanTargets, child) end
                    end
                end
            end
        end
        for i = 1, #scanTargets do
            local model = scanTargets[i]
            local ppart = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            if ppart then
                local dist = (ppart.Position - rPos).Magnitude
                if dist <= range then
                    local eid = model:GetAttribute("EntityID") or model.Parent:GetAttribute("EntityID")
                    if eid then
                        table.insert(fruitBushesCache, {entityid = eid, model = model, dist = dist, pos = ppart.Position})
                    end
                end
            end
        end
        table.sort(fruitBushesCache, function(a,b) return a.dist < b.dist end)
    end
end)

-- Character Character Position Walk Managers
local function walkToTarget(targetPosition)
    if hum and targetPosition then hum:MoveTo(targetPosition) end
end

    while true do
        task.wait(0.1)
        if Options.tweentoplantbox.Value then
            for _, box in ipairs(plantBoxesCache) do

                    walkToTarget(box.pos)

                end
            end
        elseif Options.tweentobush.Value then

            elseif #plantBoxesCache > 0 then
                for _, box in ipairs(plantBoxesCache) do
                    if box.deployable and not box.deployable:FindFirstChild("Seed") then
                        walkToTarget(box.pos)
                        break
                    end
                end
            end
        end
    end
end)

-- Highly Optimized Autoplant Thread
local plantedboxes = {}
task.spawn(function()
    while true do
        if not Options.planttoggle.Value then task.wait(0.2) continue end
        local range = tonumber(Options.plantrange.Value) or 30
        local delay = tonumber(Options.plantdelay.Value) or 0.1
        local selectedfruit = Options.fruitdropdown.Value
        local itemID = fruittoitemid[selectedfruit] or 94
        
        if root then
            local rPos = root.Position
            for i = 1, #plantBoxesCache do
                local data = plantBoxesCache[i]
                if data.dist <= range and data.deployable and data.deployable.Parent then
                    if not data.deployable:FindFirstChild("Seed") and not plantedboxes[data.entityid] then
                        plantedboxes[data.entityid] = true
                        if packets.InteractStructure and packets.InteractStructure.send then
                            packets.InteractStructure.send({ entityID = data.entityid, itemID = itemID })
                        end
                        task.spawn(function()
                            task.wait(1.0)
                            if data.deployable and not data.deployable:FindFirstChild("Seed") then
                                plantedboxes[data.entityid] = nil
                            end
                        end)
                    end
                end
            end
        end
        task.wait(delay)
    end
end)

-- Highly Optimized Auto Harvest Layer
local activelyHarvestedBushes = {}
task.spawn(function()
    while true do
        if not Options.harvesttoggle.Value then task.wait(0.1) continue end
        local hRange = tonumber(Options.harvestrange.Value) or 30
        
        for i = 1, #fruitBushesCache do
            local bush = fruitBushesCache[i]
            if bush.dist <= hRange then
                pickup(bush.entityid)
                
                if not activelyHarvestedBushes[bush.entityid] then
                    activelyHarvestedBushes[bush.entityid] = true
                    task.spawn(function()
                        local tModel = bush.model
                        while tModel and tModel.Parent do task.wait(0.1) end
                        if root and (bush.pos - root.Position).Magnitude <= hRange + 5 then
                            if not farmStats.timerStarted then
                                farmStats.timerStarted = true
                                farmStats.startTime = tick()
                                farmStats.sessionStartTime = tick()
                            end
                            farmStats.harvestedCount = farmStats.harvestedCount + 3
                            farmStats.totalAllTime = farmStats.totalAllTime + 3
                        end
                        activelyHarvestedBushes[bush.entityid] = nil
                    end)
                end
            end
        end
        task.wait(0.1)
    end
end)

-- Non-Blocking Statistics Rendering Thread (Fires every 400ms instead of frame render)
task.spawn(function()
    local lastCount = cachedInventoryCount
    while true do
        task.wait(0.4)
        local currentCount = cachedInventoryCount
        local delta = 0
        
        if currentCount and currentCount > 0 then
            if currentCount > lastCount then
                delta = currentCount - lastCount
                lastCount = currentCount
            elseif currentCount < lastCount then
                lastCount = currentCount
            end
        end
        
        if delta > 0 then
            if not farmStats.timerStarted then
                farmStats.timerStarted = true
                farmStats.startTime = tick()
                farmStats.sessionStartTime = tick()
            end
            farmStats.harvestedCount = farmStats.harvestedCount + delta
            farmStats.totalAllTime = farmStats.totalAllTime + delta
        end

        local runElapsed = farmStats.timerStarted and (tick() - farmStats.startTime) or 1
        if runElapsed < 1 then runElapsed = 1 end
        
        local itemsPerMin = farmStats.timerStarted and math.round((farmStats.harvestedCount / runElapsed) * 60) or 0
        local itemsPerHour = farmStats.timerStarted and math.round((farmStats.harvestedCount / runElapsed) * 3600) or 0
        local targetFruit = Options.fruitdropdown.Value
        
        if HarvestedStats and HarvestedStats.SetTitle then
            HarvestedStats:SetTitle(string.format("%s Harvested (Session): %d (Per Min: %d | Per Hour: %d)", targetFruit, farmStats.harvestedCount, itemsPerMin, itemsPerHour))
        end
        
        if TotalStats and TotalStats.SetTitle then
            if farmStats.timerStarted then
                local allTimeElapsed = tick() - farmStats.sessionStartTime
                TotalStats:SetTitle(string.format("Total Harvested (All Time): %d (Time: %s)", farmStats.totalAllTime, formatTime(allTimeElapsed)))
            else
                TotalStats:SetTitle(string.format("Total Harvested (All Time): %d (Time: 00:00:00)", farmStats.totalAllTime))
            end
        end
    end
end)

-- Snapped Plantbox Placement Code Engine 
function placeSingleSnappedBox()
    if not (root and workspace.CurrentCamera) then return end
    local pos = root.Position
    local gs = 5.05
    local sp = Vector3.new(math.round(pos.X/gs)*gs, pos.Y, math.round(pos.Z/gs)*gs)
    
    local rp = RaycastParams.new()
    rp.FilterType, rp.FilterDescendantsInstances, rp.IgnoreWater = Enum.RaycastFilterType.Blacklist, {char}, true
    local rr = workspace:Raycast(Vector3.new(sp.X, sp.Y+100, sp.Z), Vector3.new(0,-200,0), rp)
    local fp = rr and rr.Position or sp
    
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({buildingName = "Plant Box", cframe = CFrame.new(fp), vec = fp, isMobile = false, cursor = workspace.CurrentCamera.CFrame.Position})
    end
end

local function getInventorySlot(itemName)
    local list = plr.PlayerGui:FindFirstChild("MainGui") and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") and plr.PlayerGui.MainGui.RightPanel:FindFirstChild("Inventory") and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if not list then return nil end
    local slot = list:FindFirstChild(itemName)
    return slot and slot.LayoutOrder
end

local function brewPotions()
    if isBrewing or not root then return end
    isBrewing = true
    
    local dep = workspace:FindFirstChild("Deployables")
    if not dep then isBrewing = false; return end
    
    local cauldrons = {}
    local children = dep:GetChildren()
    for i = 1, #children do
        local obj = children[i]
        if obj.Name == "Cauldron" and obj:FindFirstChild("Hitbox") and (obj.Hitbox.Position - root.Position).Magnitude < 30 then
            table.insert(cauldrons, obj)
        end
    end
    
    if #cauldrons == 0 then isBrewing = false; return end
    local recipe = POTIONS[selectedPotion]
    if not recipe then isBrewing = false; return end
    
    for _, ing in ipairs(recipe) do
        local slot = getInventorySlot(ing.name)
        if slot then
            for _, c in ipairs(cauldrons) do
                for _ = 1, ing.amount do 
                    packets.DropBagItem.send(slot)
                    task.wait(0.05)
                end
                task.wait(0.2)
                
                local tp = c.Hitbox.Position + Vector3.new(0, 0.8, 0)
                if workspace:FindFirstChild("Items") then
                    for _, item in ipairs(workspace.Items:GetChildren()) do
                        if item:IsA("BasePart") and item.Name == ing.name then item.Position = tp end
                    end
                end
                task.wait(0.2)
            end
        end
    end
    task.wait(1.5)
    for _, c in ipairs(cauldrons) do
        local eid = c:GetAttribute("EntityID")
        if eid then packets.SweepPotion.send(eid) end
    end
    isBrewing = false
end
