local function float16(v)
    if v == 0 then return 0 end
    local sign = v < 0 and 0x8000 or 0
    v = math.abs(v)
    local exp = math.floor(math.log(v, 2))
    local mant = math.floor((v / 2 ^ exp) * 1024)
    return sign + (exp + 15) * 32 + mant
end

local function swingencode(ids, cf, ts)
    if typeof(ids) ~= "table" then ids = {ids} end
    local parts = {string.char(0x00, 0x13)}
    table.insert(parts, string.pack("<H", #ids))
    for i, v in ipairs(ids) do
        table.insert(parts, string.pack("<I4x", v))
    end
    local x, y, z = cf:ToEulerAnglesXYZ()
    table.insert(parts, string.pack("<fffHHHd", cf.X, cf.Y, cf.Z, float16(x), float16(y), float16(z), ts))
    return table.concat(parts)
end

-- RAW PICKUP (ByteNet - more undetected)
local function pickupencode(entityId)
    -- Simple & effective version (most games accept this)
    return buffer.fromstring(string.pack("<I4", entityId))
end

local function rawPickup(entityId)
    if not entityId then return end
    pcall(function()
        local packet = pickupencode(entityId)
        rs:WaitForChild("ByteNetReliable"):FireServer(packet)
    end)
end

local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

    if GameUtil and GameUtil.Data then
        local inv = GameUtil.Data.inventory or GameUtil.Data.Inventory or GameUtil.Data.bag or GameUtil.Data.Bag
        if type(inv) == "table" then

                if type(v) == "table" and (v.name == itemName or v.Name == itemName) then
                    return tonumber(v.quantity or v.amount or v.Amount or v.count or v.Count or 1) or 0
                end
            end
        end
    end

end

    while true do
        local range = tonumber(Options.pickuprange.Value) or 35
        if Options.autopickuptoggle.Value and root then
            for _, item in ipairs(workspace.Items:GetChildren()) do
                if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") then
                    if table.find(selecteditems, item.Name) then
                        if (item.Position - root.Position).Magnitude <= range then
                            rawPickup(item:GetAttribute("EntityID"))
                            task.wait(0.03) -- Anti-spam
                        end

                end
            end
        end

        if Options.chestpickuptoggle.Value and root then
            local deployablesFolder = workspace:FindFirstChild("Deployables")

                        for _, item in ipairs(chest.Contents:GetChildren()) do
                            if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then

                                    task.wait(0.05)
                                end
                            end
                        end
                    end
                end
            end
        end
