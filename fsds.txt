local function getPacketId(packetTable)
    if type(packetTable) ~= "table" then return nil end
    if packetTable._id then return packetTable._id end
    if packetTable.id then return packetTable.id end
    for _, v in pairs(packetTable) do
        if type(v) == "number" then return v end
    end
    return nil
end

local function pickupencode(entityId)
    local packetId = getPacketId(packets.Pickup) or 0xE7  -- fallback
    local header = string.char(0x00, packetId)
    local data = string.pack("<I4", entityId)
    return header .. data
end

local function silentPickup(entityId)
    if not entityId then return end
    pcall(function()
        local packet = pickupencode(entityId)
        rs:WaitForChild("ByteNetReliable"):FireServer(buffer.fromstring(packet))
    end)
end

local function float16(v)
    if v == 0 then return 0 end
    local sign = v < 0 and 0x8000 or 0
    v = math.abs(v)
    local exp = math.floor(math.log(v, 2))
    local mant = math.floor((v / 2 ^ exp) * 1024)
    return sign + (exp + 15) * 32 + mant
end

local function swingencode(ids, cf, ts)
    if typeof(ids) ~= "table" then ids = {ids} end
    local parts = {string.char(0x00, 0x13)}
    table.insert(parts, string.pack("<H", #ids))
    for _, v in ipairs(ids) do
        table.insert(parts, string.pack("<I4x", v))
    end
    local x, y, z = cf:ToEulerAnglesXYZ()
    table.insert(parts, string.pack("<fffHHHd", cf.X, cf.Y, cf.Z, float16(x), float16(y), float16(z), ts))
    return table.concat(parts)
end

    local packet = swingencode(entityIds, rootPart.CFrame, ts)
    pcall(function()
        rs:WaitForChild("ByteNetReliable"):FireServer(buffer.fromstring(packet))
    end)
end
