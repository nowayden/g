-- Helper function to format seconds into HH:MM:SS
local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- Robust inventory scanner that checks background data modules and UI layouts
local function getInventoryCount(itemName)
    if GameUtil and GameUtil.Data then
        local inv = GameUtil.Data.inventory or GameUtil.Data.Inventory or GameUtil.Data.bag or GameUtil.Data.Bag
        if type(inv) == "table" then
             if inv[itemName] then return tonumber(inv[itemName]) or 0 end
            for _, v in pairs(inv) do
                if type(v) == "table" and (v.name == itemName or v.Name == itemName) then
                    return tonumber(v.quantity or v.amount or v.Amount or v.count or v.Count or 1) or 0
               end
            end
        end
    end
    
    local list = plr.PlayerGui:FindFirstChild("MainGui") 
        and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") 
        and plr.PlayerGui:FindFirstChild("MainGui").RightPanel:FindFirstChild("Inventory") 
        and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if list then
        local total = 0
        for _, slot in ipairs(list:GetChildren()) do
            if slot:IsA("GuiObject") and (slot.Name == itemName or slot.Name:find(itemName)) then
                local quantityImg = slot:FindFirstChild("QuantityImage")
                local textObj = (quantityImg and quantityImg:FindFirstChild("QuantityText")) or slot:FindFirstChild("Amount") or slot:FindFirstChild("Count") or slot:FindFirstChildWhichIsA("TextLabel")
                if textObj then
                     total = total + (tonumber(textObj.Text:match("%d+")) or 1)
                else
                    total = total + 1
                end
            end
        end
        if total > 0 then return total end
    end
    return 0
end

-- Helper functions
local function swingtool(entityIds)
    if not (packets.SwingTool and packets.SwingTool.send) then return end
    local arr = {}
    for _, eid in ipairs(entityIds) do table.insert(arr, {entityID = eid, buffer = 0}) end
    pcall(function()
         packets.SwingTool.send({entityIDs = arr, cframe = root and root.CFrame or CFrame.new(), timestamp = clock.getServerTime(true)})
    end)
end

-- Resource Aura
local resourceActive, resourceConn, lastResourceTime = false, nil, 0
local function startResource()
    if resourceActive then return end
    resourceActive = true
    resourceConn = runs.Heartbeat:Connect(function()
         if not Options.resourceauratoggle.Value then resourceConn:Disconnect(); resourceConn = nil; resourceActive = false; return end
        if not root or not GameUtil.Data.equipped then return end
        local ct = tick()
        if ct - lastResourceTime < (tonumber(Options.resourcecooldownslider.Value) or 0.1) then return end
        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1
        local targets, all = {}, {}
        
         if workspace:FindFirstChild("Resources") then
            for _, r in pairs(workspace.Resources:GetDescendants()) do 
                if r:IsA("Model") then 
                    table.insert(all, r) 
                end 
            end
        end
 
        local resourceKeywords = {"Node", "Stone", "Lode", "Rock", "Coal", "Guardian"}
        for _, r in pairs(workspace:GetChildren()) do 
            if r:IsA("Model") then
                local matchFound = false
                for _, kw in ipairs(resourceKeywords) do
                     if string.find(r.Name, kw) then
                        matchFound = true
                        break
                    end
                end
                 if matchFound then
                    table.insert(all, r)
                end
            end 
        end
        
        for _, res in pairs(all) do
             if res:IsA("Model") and res:GetAttribute("EntityID") then
                local pp = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                if pp and (pp.Position - root.Position).Magnitude <= range then
                    table.insert(targets, {eid = res:GetAttribute("EntityID"), dist = (pp.Position - root.Position).Magnitude})
                end
             end
        end
        if #targets > 0 then
            table.sort(targets, function(a,b) return a.dist < b.dist end)
            local sel = {}
            for i = 1, math.min(tc, #targets) do table.insert(sel, targets[i].eid) end
            swingtool(sel)
             lastResourceTime = ct
        end
    end)
end
Options.resourceauratoggle:OnChanged(function(s) if s then startResource() else resourceActive = false; if resourceConn then resourceConn:Disconnect(); resourceConn = nil end end end)

-- ========= [ DEPLOYABLE AURA FROM V9 ] =========
local RS = runs
local isTraderActive = false

local deployableAuraActive = false
local deployableAuraConnection = nil
local lastDeployableSwingTime = 0

function StartDeployableAura(range)
    if deployableAuraActive then return end
    deployableAuraActive = true
    range = range or 25

    deployableAuraConnection = RS.Heartbeat:Connect(function()
        if not deployableAuraActive or isTraderActive then return end

        local currentTime = tick()
        if currentTime - lastDeployableSwingTime < 0.1 then return end

        local char = plr.Character
        local rootPart = char and char:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end

        local targetList = {}
        local folders = {"Deployables", "Rubble", "dropChests"}
      
        for _, folderName in ipairs(folders) do
            local folder = workspace:FindFirstChild(folderName)
             if folder then
                for _, item in ipairs(folder:GetChildren()) do
                    if item:IsA("Model") and item:GetAttribute("EntityID") then
                        local primaryPart = item.PrimaryPart or item:FindFirstChildWhichIsA("BasePart")
                        if primaryPart then
                            local distance = (primaryPart.Position - rootPart.Position).Magnitude
                            if distance <= range then
                                table.insert(targetList, { id = item:GetAttribute("EntityID"), dist = distance })
                            end
                        end
                    end
                end
            end
        end

        if #targetList > 0 then
            table.sort(targetList, function(a, b) return a.dist < b.dist end)
            local targetID = targetList[1].id
            swingtool({targetID})
            lastDeployableSwingTime = currentTime
        end
    end)
end

function StopDeployableAura()
    deployableAuraActive = false
    if deployableAuraConnection then
        deployableAuraConnection:Disconnect()
        deployableAuraConnection = nil
    end
end

-- Farming path helpers
local lastPos = Vector3.new()
local stuckTimer = 0
local lastCheck = 0
local function walkToTarget(targetPosition)
    if hum and root and targetPosition then
        hum:MoveTo(targetPosition)
        local now = tick()
        local dt = now - lastCheck
        if lastCheck == 0 or dt > 0.5 then
            lastCheck = now
            lastPos = root.Position
            stuckTimer = 0
            return
        end
        lastCheck = now
        if (root.Position - targetPosition).Magnitude > 3 then
            if (root.Position - lastPos).Magnitude < 1 * dt then
                stuckTimer = stuckTimer + dt
                if stuckTimer >= 3 then
                    hum.Jump = true
                    stuckTimer = 0
                end
            else
                stuckTimer = 0
            end
        else
            stuckTimer = 0
        end
        lastPos = root.Position
    end
end

local function getpbs(range)
    local arr = {}
    local deployablesFolder = workspace:FindFirstChild("Deployables")
    if deployablesFolder and root then
        for _, v in ipairs(deployablesFolder:GetChildren()) do
            if v:IsA("Model") and v.Name == "Plant Box" then
                local ppart = v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")
                local entityid = v:GetAttribute("EntityID")
                if ppart and entityid then
                    local dist = (ppart.Position - root.Position).Magnitude
                    if dist <= range then
                        table.insert(arr, {deployable = v, part = ppart, id = entityid, dist = dist})
                    end
                end
            end
        end
    end
    return arr
end

local function getbushes(range, fruitName)
    local arr = {}
    local bushesFolder = workspace:FindFirstChild("Bushes") or workspace:FindFirstChild("Bushes")
    if bushesFolder and root then
        for _, v in ipairs(bushesFolder:GetChildren()) do
            if v:IsA("Model") and v.Name:find("Bush") then
                for _, child in ipairs(v:GetChildren()) do
                    if child:IsA("Model") and child.Name == fruitName then
                        local ppart = child.PrimaryPart or child:FindFirstChildWhichIsA("BasePart")
                        local entityid = child:GetAttribute("EntityID")
                        if ppart and entityid then
                            local dist = (ppart.Position - root.Position).Magnitude
                            if dist <= range then
                                table.insert(arr, {model = child, part = ppart, entityid = entityid, dist = dist})
                            end
                        end
                    end
                end
            end
        end
    end
    return arr
end

local function pathToPlantbox(range)
    local plantboxes = getpbs(range)
    table.sort(plantboxes, function(a, b) return a.dist < b.dist end)
    for _, box in ipairs(plantboxes) do
        if not box.deployable:FindFirstChild("Seed") then
            if box.deployable:GetAttribute("Growth") and box.deployable:GetAttribute("Growth") < 100 then
                walkToTarget(box.part.Position)
                return true
            end
        end
    end
    return false
end

local function pathToBushesOrBoxes(range, fruitName)
    local bushes = getbushes(range, fruitName)
    local boxes = getpbs(range)
    local targets = {}
    
    for _, b in ipairs(bushes) do table.insert(targets, {part = b.part, dist = b.dist}) end
    for _, box in ipairs(boxes) do
        if box.deployable:FindFirstChild("Seed") and box.deployable:GetAttribute("Growth") == 100 then
            table.insert(targets, {part = box.part, dist = box.dist})
        end
    end
    
    if #targets > 0 then
        table.sort(targets, function(a, b) return a.dist < b.dist end)
        walkToTarget(targets[1].part.Position)
        return true
    end
    return false
end

task.spawn(function()

            local range = tonumber(Options.tweenrange.Value) or 250

local plantedboxes = {}
local plantCache = {}
local lastPlantCacheUpdate = 0

local function getInventorySlot(itemName)
    local list = plr.PlayerGui:FindFirstChild("MainGui") and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") and plr.PlayerGui.MainGui.RightPanel:FindFirstChild("Inventory") and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if not list then return nil end
    for _, slot in ipairs(list:GetChildren()) do
        if slot:IsA("ImageLabel") and slot.Name:find(itemName) then
            return slot.LayoutOrder
        end
    end
end

local function plant(boxEntityID, itemID)
    if plantedboxes[boxEntityID] then return end
    if packets.PlantSeed and packets.PlantSeed.send then
        plantedboxes[boxEntityID] = true
        packets.PlantSeed.send({entityID = boxEntityID, slot = itemID})
        task.wait(0.01)
        plantedboxes[boxEntityID] = nil
    end
end

task.spawn(function()
    while true do
        if not Options.planttoggle.Value then task.wait(0.1) continue end
        local seedName = fruittoitemid[Options.fruitdropdown.Value]
        local itemID = getInventorySlot(seedName)
        if not itemID then task.wait(0.2) continue end
        
        local range = tonumber(Options.plantrange.Value) or 30
        local delay = tonumber(Options.plantdelay.Value) or 0.1
        
        local currentTime = tick()
        local deployablesFolder = workspace:FindFirstChild("Deployables")
        if deployablesFolder and (currentTime - lastPlantCacheUpdate > 0.5) then
            plantCache = {}
            for _, v in ipairs(deployablesFolder:GetChildren()) do
                if v:IsA("Model") and v.Name == "Plant Box" then
                    local ppart = v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")
                    local entityid = v:GetAttribute("EntityID")
                    if ppart and entityid then
                        table.insert(plantCache, {box = v, part = ppart, id = entityid})
                    end
                end
            end
            lastPlantCacheUpdate = currentTime
        end
        
        if root then
            local myPos = root.Position
            for i = 1, #plantCache do
                local data = plantCache[i]
                if data.box.Parent == deployablesFolder then
                    if (data.part.Position - myPos).Magnitude <= range then
                        if not data.box:FindFirstChild("Seed") then
                            plant(data.id, itemID)
                        else
                            plantedboxes[data.id] = true
                        end
                    end
                end
            end
        end
        task.wait(delay)
    end
end)

    end
end)

    if child:IsA("Model") and child.Name == "Bloodfruit" then

-- Plant box placement (Tight Edge-to-Edge Snapping)
function placeSingleSnappedBox()
    local char = plr.Character
    if not (char and char:FindFirstChild("HumanoidRootPart")) then return end
    local pos = char.HumanoidRootPart.Position
    local cam = workspace.CurrentCamera
    local cursorPos = cam and cam.CFrame.Position or pos
    local gs = 5.05
    local sp = Vector3.new(math.round(pos.X/gs)*gs, pos.Y, math.round(pos.Z/gs)*gs)
    local rp = RaycastParams.new()
    rp.FilterType, rp.FilterDescendantsInstances, rp.IgnoreWater = Enum.RaycastFilterType.Blacklist, {char}, true
    local rr = workspace:Raycast(Vector3.new(sp.X, sp.Y+100, sp.Z), Vector3.new(0,-200,0), rp)
    local fp = rr and rr.Position or sp
    local rot = CFrame.Angles(0, 0, 0)
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({buildingName = "Plant Box", cframe = CFrame.new(fp)*rot, vec = fp, isMobile = false, cursor = cursorPos})
    end
end

local placementConn
Options.placeplantboxestoggle:OnChanged(function(v)
    if v then
        if placementConn then placementConn:Disconnect() end
        placementConn = runs.Heartbeat:Connect(function()
            placeSingleSnappedBox();
            task.wait(0.1)
        end)
    else
        if placementConn then placementConn:Disconnect(); placementConn = nil end
    end
end)

-- Precise Forward-Snapped Dock Placement Logic (Independent Axes Handling)
function placeSingleDock()
    local char = plr.Character
    if not (char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChild("Humanoid")) then return end
    
    local rootPart = char.HumanoidRootPart
    local moveDir = char.Humanoid.MoveDirection
    local lookDir = rootPart.CFrame.LookVector
    
    -- Pick direction: use running direction if moving, fall back to look vector if standing still
    local forwardDir = moveDir.Magnitude > 0.1 and moveDir.Unit or lookDir
    forwardDir = Vector3.new(forwardDir.X, 0, forwardDir.Z).Unit -- Flatten height variations
    
    -- Adjust how far ahead the structure spawns so you run onto it perfectly
    local offsetDistance = Options.dockoffsetslider and Options.dockoffsetslider.Value or 14
    local targetPos = rootPart.Position + (forwardDir * offsetDistance)
    
    -- Split axes snapping allows you to configure rectangular structures cleanly
    local gsX = Options.dockgridXslider and Options.dockgridXslider.Value or 15.0
    local gsZ = Options.dockgridZslider and Options.dockgridZslider.Value or 5.05
    local sp = Vector3.new(math.round(targetPos.X / gsX) * gsX, rootPart.Position.Y, math.round(targetPos.Z / gsZ) * gsZ)
    
    local cam = workspace.CurrentCamera
    local cursorPos = cam and cam.CFrame.Position or rootPart.Position
    
    -- Include water layer targets to position the structure levelly over sea surfaces
    local rp = RaycastParams.new()
    rp.FilterType, rp.FilterDescendantsInstances, rp.IgnoreWater = Enum.RaycastFilterType.Blacklist, {char}, false
    
    local rr = workspace:Raycast(Vector3.new(sp.X, sp.Y + 100, sp.Z), Vector3.new(0, -200, 0), rp)
    local fp = rr and rr.Position or Vector3.new(sp.X, rootPart.Position.Y - 2.8, sp.Z)
    
    local rot = CFrame.Angles(0, 0, 0)
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({
            buildingName = "Dock",
            cframe = CFrame.new(fp) * rot,
            vec = fp,
            isMobile = false,
            cursor = cursorPos
        })
    end
end

local dockPlacementConn
local lastDockPlace = 0
Options.placedockstoggle:OnChanged(function(v)
    if v then
        if dockPlacementConn then dockPlacementConn:Disconnect() end
        dockPlacementConn = runs.Heartbeat:Connect(function()
            local now = tick()
            if now - lastDockPlace >= 0.12 then -- Smooth continuous deployment rate limiting
                lastDockPlace = now
                placeSingleDock()
            end
        end)
    else
        if dockPlacementConn then dockPlacementConn:Disconnect(); dockPlacementConn = nil end
    end
end)

local function brewPotions()
    if isBrewing then return end
    local recipe = POTIONS[selectedPotion]
    if not recipe then return end
    
    for _, ingredient in ipairs(recipe) do
        if getInventoryCount(ingredient.name) < ingredient.amount then
            return
        end
    end
    
    local cauldron = nil
    local deployables = workspace:FindFirstChild("Deployables")
    if deployables then
        for _, d in ipairs(deployables:GetChildren()) do
            if d.Name == "Cauldron" and d:GetAttribute("EntityID") then
                local p = d.PrimaryPart or d:FindFirstChildWhichIsA("BasePart")
                if p and root and (p.Position - root.Position).Magnitude <= 20 then
                    cauldron = d
                    break
                end
            end
        end
    end
    
    if not cauldron then return end
    isBrewing = true
    
    task.spawn(function()
        for _, ingredient in ipairs(recipe) do
            local slot = getInventorySlot(ingredient.name)
            if slot then
                for i = 1, ingredient.amount do
                    if packets.UseCauldron and packets.UseCauldron.send then
                        packets.UseCauldron.send({cauldronID = cauldron:GetAttribute("EntityID"), slot = slot})
                        task.wait(0.05)
                    end
                end
            end
        end
        if packets.TakePotion and packets.TakePotion.send then
            packets.TakePotion.send(cauldron:GetAttribute("EntityID"))
        end
        task.wait(0.2)
        isBrewing = false
    end)
end

-- Targeted Priorities Parser for Boss Tracking
local function getPriorityTarget()
    local char = plr.Character
    local rootPart = char and char:FindFirstChild("HumanoidRootPart")
    if not rootPart then return nil end

    local crittersFolder = workspace:FindFirstChild("Critters")
    local deployablesFolder = workspace:FindFirstChild("Deployables")
    
    local traderInstance = nil
    if deployablesFolder then
        traderInstance = deployablesFolder:FindFirstChild("Mysterious Trader")
    end

    local queenInstance = nil
    local minionInstance = nil
    local antInstance = nil

    local queenMinDist = math.huge
    local minionMinDist = math.huge
    local antMinDist = math.huge

    if crittersFolder then
        for _, c in ipairs(crittersFolder:GetChildren()) do
            if c:IsA("Model") and c:GetAttribute("EntityID") then
                local primaryPart = c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart")
                if primaryPart then
                    local dist = (primaryPart.Position - rootPart.Position).Magnitude
                    
                    -- Priority 1: Voodoo Spider Queen
                    if c.Name == "Voodoo Spider Queen" and not traderInstance then
                        if dist < queenMinDist then
                            queenMinDist = dist
                            queenInstance = c
                        end
                    -- Priority 2: Soul Minion
                    elseif c.Name == "Soul Minion" and not traderInstance and not queenInstance then
                        if dist < minionMinDist then
                            minionMinDist = dist
                            minionInstance = c
                        end
                    -- Priority 3: Void Ant
                    elseif c.Name == "Void Ant" and not traderInstance and not queenInstance and not minionInstance then
                        if dist < antMinDist then
                            antMinDist = dist
                            antInstance = c
                        end
                    end
                end
            end
        end
    end

    if traderInstance then return traderInstance end
    if queenInstance then return queenInstance end
    if minionInstance then return minionInstance end
    if antInstance then return antInstance end
    return nil
end

local queenServantActive, queenServantConn, lastQueenServantTime = false, nil, 0
local function startQueenServantAura()
    if queenServantActive then return end
    queenServantActive = true
    queenServantConn = runs.Heartbeat:Connect(function()
        if not Options.queenservanttoggle.Value then queenServantConn:Disconnect(); queenServantConn = nil; queenServantActive = false; return end
        if not root or not GameUtil.Data.equipped then return end
        local ct = tick()
        if ct - lastQueenServantTime < 0.1 then return end
        
        local target = getPriorityTarget()
        if target then
            local entityID = target:GetAttribute("EntityID")
            local isSoulMinion = (target.Name == "Soul Minion")
            
            if Options.tpsoulminionstoggle.Value and isSoulMinion and target.PrimaryPart then
                pcall(function()
                    root.CFrame = target.PrimaryPart.CFrame * CFrame.new(0, 0, 3)
                end)
            end
            
            if entityID then
                swingtool({entityID})
                lastQueenServantTime = ct
            end
        end
    end)
end

local function stopQueenServantAura()
    queenServantActive = false
    if queenServantConn then queenServantConn:Disconnect(); queenServantConn = nil end
end
