task.spawn(function()
    while true do
        if not Options.resourceauratoggle.Value then

        end
        local range = tonumber(Options.resourceaurarange.Value) or 20
        local targetCount = tonumber(Options.resourcetargetdropdown.Value) or 1

        local targets = {}
        local allresources = {}
        for _, r in pairs(workspace.Resources:GetChildren()) do
            table.insert(allresources, r)
        end
        for _, r in pairs(workspace:GetChildren()) do
            if r:IsA("Model") and r.Name == "Gold Node" then

            end
        end
        for _, res in pairs(allresources) do
            if res:IsA("Model") and res:GetAttribute("EntityID") then
                local eid = res:GetAttribute("EntityID")
                local ppart = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                if ppart then
                    local dist = (ppart.Position - root.Position).Magnitude
                    if dist <= range then
                        table.insert(targets, { eid = eid, dist = dist })
                    end
                end
            end
        end
        if #targets > 0 then
            table.sort(targets, function(a, b)
                return a.dist < b.dist
            end)
            local selectedTargets = {}
            for i = 1, math.min(targetCount, #targets) do
                table.insert(selectedTargets, targets[i].eid)
            end
            swingtool(selectedTargets)
        end
        task.wait(cooldown)
    end
end)

local function getpbs(range)
    local plantboxes = {}
    for _, deployable in ipairs(workspace.Deployables:GetChildren()) do
        if deployable:IsA("Model") and deployable.Name == "Plant Box" then
            local entityid = deployable:GetAttribute("EntityID")
            local ppart = deployable.PrimaryPart or deployable:FindFirstChildWhichIsA("BasePart")
            -- Only add if root exists to prevent 'Position' error
            if entityid and ppart and root then 
                local dist = (ppart.Position - root.Position).Magnitude
                if dist <= range then
                    table.insert(plantboxes, { entityid = entityid, deployable = deployable, dist = dist })
                end
            end
        end
    end
    return plantboxes
end

local function getbushes(range, fruitname)
    local bushes = {}
    for _, model in ipairs(workspace:GetChildren()) do
        if model:IsA("Model") and model.Name:find(fruitname) then
            local ppart = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            -- Only add if root exists to prevent 'Position' error
            if ppart and root then 
                local dist = (ppart.Position - root.Position).Magnitude
                if dist <= range then
                    local entityid = model:GetAttribute("EntityID")
                    if entityid then
                        table.insert(bushes, { entityid = entityid, model = model, dist = dist })
                    end
                end
            end
        end
    end
    return bushes
end

-- TWEENING SYSTEM WITH SPEED CONTROL
local tweening = nil
local lastPlantTime = 0
local function tween(target)
    if not root then return end -- Check if root exists before tweening
    if tweening then 
        tweening:Cancel()
        tweening = nil
    end
    
    local distance = (root.Position - target.Position).Magnitude
    -- USE SPEED SLIDER: Higher value = Faster movement
    local tweenSpeed = Options.tweenspeed.Value
    local duration = distance / tweenSpeed  -- Speed controlled by slider
    
    -- Safety: Minimum duration to prevent instant teleport at very high speeds
    if duration < 0.1 then duration = 0.1 end
    
    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    local tween = tspmo:Create(root, tweenInfo, { CFrame = target })
    tween:Play()
    tweening = tween
    return tween
end

-- IMPROVED: TWEEN TO PLANT BOX WITH SPEED CONTROL AND SMOOTHNESS
local function tweenplantbox(range)
    if not root then task.wait(0.1); return end -- Early return if no root
    -- Get the list of available plant boxes within range that don't have seeds
    local plantboxes = getpbs(range)
    local closestEmptyBox = nil
    local minDist = math.huge

    for _, box in ipairs(plantboxes) do
        if not box.deployable:FindFirstChild("Seed") then
            if box.dist < minDist then
                minDist = box.dist
                closestEmptyBox = box
            end
        end
    end

    if closestEmptyBox then
        local targetCFrame = closestEmptyBox.deployable.PrimaryPart.CFrame + Vector3.new(0, 3, 0) -- Adjust offset as needed
        -- Call the updated tween function which cancels previous tweens
        local currentTween = tween(targetCFrame)
    end
end

-- IMPROVED: TWEEN TO BUSH OR PLANT BOX WITH SPEED CONTROL AND SMOOTHNESS
local function tweenpbs(range, fruitname)
    if not root then task.wait(0.1); return end -- Early return if no root
    -- Get the currently selected fruit from the dropdown inside the loop
    local currentFruitName = Options.fruitdropdown.Value

    -- First, look for harvestable bushes of the specified fruit
    local bushes = getbushes(range, currentFruitName)
    local closestBush = nil
    local minDistBush = math.huge

    for _, bush in ipairs(bushes) do
        if bush.dist < minDistBush then
            minDistBush = bush.dist
            closestBush = bush
        end
    end

    if closestBush then
        -- Found a bush, move towards it
        local targetCFrame = closestBush.model.PrimaryPart.CFrame + Vector3.new(0, 3, 0) -- Adjust offset as needed
        local currentTween = tween(targetCFrame)

        -- Optional: Attempt to harvest if close enough after waiting
        -- Calculate distance again after waiting
        local newDist = (closestBush.model.PrimaryPart.Position - root.Position).Magnitude
        if newDist <= range then -- Check if still within range
             -- Trigger pickup for the bush
            if packets.Pickup and packets.Pickup.send then
                packets.Pickup.send(closestBush.entityid)
            end
        end

    else
        -- No bushes found, look for empty plant boxes instead
        local plantboxes = getpbs(range)
        local closestEmptyBox = nil
        local minDistBox = math.huge

        for _, box in ipairs(plantboxes) do
            if not box.deployable:FindFirstChild("Seed") then
                if box.dist < minDistBox then
                    minDistBox = box.dist
                    closestEmptyBox = box
                end
            end
        end

        if closestEmptyBox then
            -- Found an empty plant box, move towards it
            local targetCFrame = closestEmptyBox.deployable.PrimaryPart.CFrame + Vector3.new(0, 3, 0) -- Adjust offset as needed
            local currentTween = tween(targetCFrame)

             -- Optional: Attempt to plant if close enough after waiting and has seeds
             -- Calculate distance again after waiting
             local newDistToBox = (closestEmptyBox.deployable.PrimaryPart.Position - root.Position).Magnitude
             if newDistToBox <= range then -- Check if still within range
                 -- Check if player has the required seed in inventory (this part might need adjustment based on how inventory works)
                 -- For now, just attempt to plant if close enough and it's an empty box
                 -- A real check would involve verifying the player's inventory contains the specific fruit item ID
                 local itemID = fruittoitemid[currentFruitName] or 94
                 -- Simple attempt to plant (requires player to have the item)
                 if packets.InteractStructure and packets.InteractStructure.send then
                     pcall(function()
                         packets.InteractStructure.send({
                             entityID = closestEmptyBox.entityid,
                             itemID = itemID
                         })
                     end)
                     -- lastPlantTime = tick() -- Uncomment if you want a cooldown per plant action
                 end
             end


        end
    end
end

-- IMPROVED: Place Structure (Uses Snapping Logic, Handles Single and Grid Placement, Includes Continuous Option, Adds Cursor, Adds Rotation Choice)
local placementConnection = nil
local function placeSingleSnappedBox()
    local player = game.Players.LocalPlayer
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then
        warn("Cannot place box: no character or HumanoidRootPart")
        return
    end

    local rootPart = character.HumanoidRootPart
    local currentPosition = rootPart.Position
    local camera = workspace.CurrentCamera
    local cursorPos = camera and camera.CFrame.Position or currentPosition -- Fallback to root pos if camera fails

    -- Snap to grid
    local gridSize = 6.05
    local snappedX = math.round(currentPosition.X / gridSize) * gridSize
    local snappedZ = math.round(currentPosition.Z / gridSize) * gridSize
    local snappedPosition = Vector3.new(snappedX, currentPosition.Y, snappedZ)

    -- Raycast down
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Blacklist
    rayParams.FilterDescendantsInstances = {character}
    rayParams.IgnoreWater = true

    local rayOrigin = Vector3.new(snappedPosition.X, snappedPosition.Y + 100, snappedPosition.Z)
    local rayDirection = Vector3.new(0, -200, 0)
    local rayResult = workspace:Raycast(rayOrigin, rayDirection, rayParams)

    local finalPosition = rayResult and rayResult.Position or snappedPosition

    -- Determine rotation based on dropdown
    local selectedRotation = Options.rotationdropdown.Value
    local rotationCFrame = CFrame.Angles(0, 0, 0) -- 0 degrees
    if selectedRotation == "45 degrees" then
        rotationCFrame = CFrame.Angles(0, math.rad(45), 0) -- 45 degrees around Y
    end
    local finalCFrame = CFrame.new(finalPosition) * rotationCFrame

    -- ✅ CRITICAL: Include 'cursor' field (required by ByteNet) and rotation via CFrame
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({
            buildingName = "Plant Box",
            cframe = finalCFrame, -- ✅ Rotation baked in
            vec = finalPosition,
            isMobile = false,
            cursor = cursorPos  -- 👈 MUST be present and non-nil
        })
        print("Placed plant box at:", finalPosition, "with rotation:", selectedRotation)
    else
        warn("PlaceStructure packet not found or send method missing.")
    end
end

-- Grid placement function (keeps original grid logic but uses snapped positions, adds cursor, adds rotation choice)
local function placeGridOfBoxes(gridSize)
    local player = game.Players.LocalPlayer
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then
        print("Player character or root part not found for placement.")
        return
    end

    local torso = character.HumanoidRootPart
    local camera = workspace.CurrentCamera
    local cursorPos = camera and camera.CFrame.Position or torso.Position -- Fallback

    -- Determine rotation based on dropdown
    local selectedRotation = Options.rotationdropdown.Value
    local rotationCFrame = CFrame.Angles(0, 0, 0)
    if selectedRotation == "45 degrees" then
        rotationCFrame = CFrame.Angles(0, math.rad(45), 0)
    end

    local startpos = torso.Position - Vector3.new(0, 3, 0) -- Offset down slightly from player
    local spacing = 6.05 -- Approximate spacing between plant boxes (matches snap grid)

    for x = 0, gridSize - 1 do
        for z = 0, gridSize - 1 do
            task.wait(0.3) -- Add a small delay between placement attempts

            -- Calculate grid position
            local gridX = startpos.X + (x * spacing)
            local gridZ = startpos.Z + (z * spacing)
            local gridPos = Vector3.new(gridX, startpos.Y, gridZ)

             -- Snap grid position to the same grid system
            local snappedX = math.round(gridPos.X / spacing) * spacing
            local snappedZ = math.round(gridPos.Z / spacing) * spacing
            local snappedGridPos = Vector3.new(snappedX, gridPos.Y, snappedZ)

            -- Raycast down from the snapped grid position
            local rayParams = RaycastParams.new()
            rayParams.FilterType = Enum.RaycastFilterType.Blacklist
            rayParams.FilterDescendantsInstances = {character}
            rayParams.IgnoreWater = true

            local rayOrigin = Vector3.new(snappedGridPos.X, snappedGridPos.Y + 100, snappedGridPos.Z)
            local rayDirection = Vector3.new(0, -200, 0)

            local rayResult = workspace:Raycast(rayOrigin, rayDirection, rayParams)

            local finalPosition = rayResult and rayResult.Position or snappedGridPos
            local finalCFrame = CFrame.new(finalPosition) * rotationCFrame -- ✅ Rotated CFrame

            -- ✅ CRITICAL: Include 'cursor' field (required by ByteNet) and rotation via CFrame
            if packets.PlaceStructure and packets.PlaceStructure.send then
                packets.PlaceStructure.send({
                    buildingName = "Plant Box",
                    cframe = finalCFrame, -- ✅ Rotation baked in
                    vec = finalPosition,
                    isMobile = false,
                    cursor = cursorPos -- 👈 MUST be present and non-nil
                })
                print("Placed plant box in grid at:", finalPosition, "with rotation:", selectedRotation)
            else
                 warn("PlaceStructure packet not found or send method missing during grid placement.")
                 break -- Stop if packet fails
            end
        end
    end
end

-- Update the placestructure function to handle both grid and single snapped placement
placestructure = function(gridsize)
    if gridsize == 1 then
        -- If gridsize is 1, treat it as a single snapped placement
        placeSingleSnappedBox()
    else
        -- Otherwise, treat it as a grid placement
        placeGridOfBoxes(gridsize)
    end
end

-- Handle the continuous placement toggle
Options.placeplantboxestoggle:OnChanged(function(value)
    if value then
        -- Enable continuous placement
        if placementConnection then
            placementConnection:Disconnect() -- Disconnect any existing connection
        end
        placementConnection = runs.Heartbeat:Connect(function()
            placeSingleSnappedBox() -- Place a box every heartbeat frame
            task.wait(0.1) -- Add a small delay to prevent spamming too quickly
        end)
    else
        -- Disable continuous placement
        if placementConnection then
            placementConnection:Disconnect()
            placementConnection = nil
        end
    end
end)
