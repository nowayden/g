    end

-- OPTIMIZED: Cached plant box finding
local function getpbs(range)
    local now = tick()
    if now - _getpbs_cache.timestamp < CACHE_DURATION then
        return _getpbs_cache.data
    end
    
    local plantboxes = {}
    local deployables = workspace.Deployables:GetChildren()
    
    for i = 1, #deployables do
        local deployable = deployables[i]
        if deployable.Name == "Plant Box" then
            local entityid = deployable:GetAttribute("EntityID")
            local ppart = deployable.PrimaryPart
            if entityid and ppart then
                local dist = (ppart.Position - root.Position).Magnitude
                if dist <= range then
                    table.insert(plantboxes, { 
                        entityid = entityid, 
                        deployable = deployable, 
                        dist = dist,
                        hasSeed = deployable:FindFirstChild("Seed") ~= nil
                    })
                end
            end
        end
    end
    
    -- Sort by distance
    table.sort(plantboxes, function(a, b) return a.dist < b.dist end)
    
    _getpbs_cache.timestamp = now
    _getpbs_cache.data = plantboxes
    
    return plantboxes
end

-- OPTIMIZED: Cached bush finding
local function getbushes(range, fruitname)
    local now = tick()
    if now - _getbushes_cache.timestamp < CACHE_DURATION and _getbushes_cache.fruitname == fruitname then
        return _getbushes_cache.data
    end
    
    local bushes = {}
    local workspaceChildren = workspace:GetChildren()
    
    for i = 1, #workspaceChildren do
        local model = workspaceChildren[i]
        if model:IsA("Model") and string.find(model.Name:lower(), fruitname:lower()) then
            local ppart = model.PrimaryPart
            if ppart then
                local dist = (ppart.Position - root.Position).Magnitude
                if dist <= range then
                    local entityid = model:GetAttribute("EntityID")
                    if entityid then
                        table.insert(bushes, { 
                            entityid = entityid, 
                            model = model, 
                            dist = dist 
                        })
                    end
                end
            end
        end
    end
    
    -- Sort by distance
    table.sort(bushes, function(a, b) return a.dist < b.dist end)
    
    _getbushes_cache.timestamp = now
    _getbushes_cache.data = bushes
    _getbushes_cache.fruitname = fruitname
    
    return bushes
end

-- ULTRA FAST TELEPORT SYSTEM (No tweening, instant movement)
local function teleportTo(position)
    if root and hum then
        -- Instant teleport with humanoid state preservation
        local currentState = hum:GetState()
        hum:ChangeState(Enum.HumanoidStateType.Jumping)
        root.CFrame = CFrame.new(position) + Vector3.new(0, 3, 0)
        runs.Heartbeat:Wait() -- Wait one frame
        hum:ChangeState(currentState)
        return true
    end
    return false
end

-- ULTRA FAST TELEPORT TO PLANT BOX
local function ultraFastTeleportPlantBox(range)
    while Options.tweentoplantbox.Value do
        local plantboxes = getpbs(range)
        
        for _, box in ipairs(plantboxes) do
            if not Options.tweentoplantbox.Value then break end
            
            if not box.hasSeed then
                local targetPosition = box.deployable.PrimaryPart.Position + Vector3.new(0, 3, 0)
                
                -- INSTANT TELEPORT
                teleportTo(targetPosition)
                
                -- Optional: Face the plant box
                root.CFrame = CFrame.new(root.Position, box.deployable.PrimaryPart.Position)
                
                -- Quick plant if auto plant is enabled
                if Options.planttoggle.Value then
                    local selectedfruit = Options.fruitdropdown.Value
                    local itemID = fruittoitemid[selectedfruit] or 94
                    plant(box.entityid, itemID)
                end
                
                break -- Teleport to one at a time
            end
        end
        
        if not Options.tweentoplantbox.Value then break end
        task.wait(0.1) -- Fast check rate
    end
end

-- ULTRA FAST TELEPORT TO BUSH + PLANT BOX
local function ultraFastTeleportBush(range, fruitname)
    while Options.tweentobush.Value do
        local fruitname = Options.fruitdropdown.Value
        
        -- First try to find bushes
        local bushes = getbushes(range, fruitname)
        
        if #bushes > 0 then
            for _, bush in ipairs(bushes) do
                if not Options.tweentobush.Value then break end
                
                local targetPosition = bush.model.PrimaryPart.Position + Vector3.new(0, 3, 0)
                
                -- INSTANT TELEPORT
                teleportTo(targetPosition)
                
                -- Face the bush
                root.CFrame = CFrame.new(root.Position, bush.model.PrimaryPart.Position)
                
                -- Plant in the bush
                if packets.InteractStructure and packets.InteractStructure.send then
                    pcall(function()
                        packets.InteractStructure.send({
                            entityID = bush.entityid,
                            itemID = fruittoitemid[fruitname] or 94
                        })
                    end)
                end
                
                break -- Process one bush at a time
            end
        else
            -- If no bushes, find empty plant boxes
            local plantboxes = getpbs(range)
            
            for _, box in ipairs(plantboxes) do
                if not Options.tweentobush.Value then break end
                
                if not box.hasSeed then

                    -- INSTANT TELEPORT
                    teleportTo(targetPosition)
                    
                    -- Face the plant box

                    -- Plant in the box

                            packets.InteractStructure.send({
                                entityID = box.entityid,

-- ULTRA FAST PLACE STRUCTURE
placestructure = function(gridsize)
    if not plr or not plr.Character then return end
    local torso = plr.Character:FindFirstChild("HumanoidRootPart")
    if not torso then return end
    
    local startpos = torso.Position - Vector3.new(0, 3, 0)
    local spacing = 6.04
    
    task.spawn(function()
        for x = 0, gridsize - 1 do
            for z = 0, gridsize - 1 do
                local position = startpos + Vector3.new(x * spacing, 0, z * spacing)
                
                if packets.PlaceStructure and packets.PlaceStructure.send then
                    -- Send without waiting
                    pcall(function()
                        packets.PlaceStructure.send{
                            ["buildingName"] = "Plant Box",
                            ["yrot"] = 45,
                            ["vec"] = position,
                            ["isMobile"] = false
                        }
                    end)
                    
                    -- Very small delay for stability
                    if (x * gridsize + z) % 5 == 0 then
                        runs.Heartbeat:Wait()
                    end
                end
            end
        end
    end)
end
