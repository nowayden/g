-- ========= [ 1. HELPERS TAB (Campfire) ] =========

local function GetNearestCampfire()
    local target = nil
    local dist = 30
    for _, v in pairs(workspace.Deployables:GetChildren()) do
        if v.Name == "Campfire" then
            local mag = (plr.Character.HumanoidRootPart.Position - v:GetPivot().Position).Magnitude
            if mag < dist then dist = mag; target = v end
        end
    end
    return target
end

Tabs.Helpers:AddDropdown("CampfireWaste", {
    Title = "Select Campfire Waste",
    Values = {"Wood", "Leaves", "Log", "Stick", "Coal"},
    Default = "Wood",
    Callback = function(v) _G.CampfireWasteID = itemIDs[v] end
})

Tabs.Helpers:AddToggle("AutoFillToggle", {
    Title = "Auto Fill Campfire",
    Default = false,
    Callback = function(state)
        _G.AutoFillCampfire = state
        task.spawn(function()
            while _G.AutoFillCampfire do
                local cf = GetNearestCampfire()
                if cf and _G.CampfireWasteID then
                    packets.InteractStructure.send({
                        entityID = cf:GetAttribute("EntityID"),
                        itemID = _G.CampfireWasteID
                    })
                end
                task.wait(0.5)
            end
        end)
    end
})

-- ========= [ 2. DEN FARM TAB (Path Walking) ] =========

local function DecodeASCII(raw)
    local chars = {}
    for code in string.gmatch(raw, "(%d+)") do
        table.insert(chars, string.char(tonumber(code)))
    end
    return table.concat(chars)
end

local function LoadDenConfig()
    local success, raw = pcall(function() return readfile("Den.json") end)
    if not success then Library:Notify({Title = "Error", Content = "Den.json not found!"}) return end

    local decoded = DecodeASCII(raw)
    local ok, data = pcall(function() return httpservice:JSONDecode(decoded) end)
    if not ok or not data.positions then return end

    local vFolder = workspace:FindFirstChild("WalkerSpheres") or Instance.new("Folder", workspace)
    vFolder.Name = "WalkerSpheres"
    vFolder:ClearAllChildren()

    _G.walkPts = {}
    for i, pStr in ipairs(data.positions) do
        local c = string.split(pStr, ",")
        local vec = Vector3.new(tonumber(c[1]), tonumber(c[2]), tonumber(c[3]))
        table.insert(_G.walkPts, vec)
        
        -- Visual sphere
        local p = Instance.new("Part", vFolder)
        p.Shape = Enum.PartType.Ball; p.Size = Vector3.new(2,2,2); p.Position = vec
        p.Anchored = true; p.CanCollide = false; p.Transparency = 0.5; p.Color = Color3.new(0,1,0)
    end
    Library:Notify({Title = "Success", Content = "Loaded " .. #_G.walkPts .. " points."})
end

Tabs.DenFarm:AddButton({
    Title = "Load Den Config",
    Description = "Reads Den.json and creates visual path",
    Callback = LoadDenConfig
})

Tabs.DenFarm:AddToggle("DenWalkToggle", {
    Title = "Start Path Walking",
    Default = false,
    Callback = function(state)
        _G.GoldFarmRunning = state
        if state then
            task.spawn(function()
                while _G.GoldFarmRunning do
                    for _, pos in ipairs(_G.walkPts or {}) do
                        if not _G.GoldFarmRunning then break end
                        local hum = plr.Character.Humanoid
                        local root = plr.Character.HumanoidRootPart
                        hum:MoveTo(pos)
                        local start = tick()
                        while (root.Position - pos).Magnitude > 3 and (tick()-start) < 10 do
                            task.wait(0.3)
                            if root.Velocity.Magnitude < 0.5 then hum.Jump = true end -- Anti-stuck
                        end
                    end
                end
            end)
        end
    end
})
