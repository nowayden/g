-- Helper functions
local function swingtool(entityIds)
    if not (packets.SwingTool and packets.SwingTool.send) then return end
    local arr = {}
    for _, eid in ipairs(entityIds) do table.insert(arr, {entityID = eid, buffer = 0}) end
    pcall(function()
        packets.SwingTool.send({entityIDs = arr, cframe = root and root.CFrame or CFrame.new(), timestamp = clock.getServerTime(true)})
    end)
end

local function pickup(entityid)
    if packets.Pickup and packets.Pickup.send then packets.Pickup.send(entityid) end
end

local function drop(itemname)
    local inv = plr.PlayerGui:FindFirstChild("MainGui") and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") and plr.PlayerGui.MainGui.RightPanel:FindFirstChild("Inventory") and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if not inv then return end
    for _, child in ipairs(inv:GetChildren()) do
        if child:IsA("ImageLabel") and child.Name == itemname then
            if packets.DropBagItem and packets.DropBagItem.send then
                packets.DropBagItem.send(child.LayoutOrder)
            end
        end
    end
end

-- Resource Aura
local resourceActive, resourceConn, lastResourceTime = false, nil, 0
local function startResource()
    if resourceActive then return end
    resourceActive = true
    resourceConn = runs.Heartbeat:Connect(function()
        if not Options.resourceauratoggle.Value then resourceConn:Disconnect(); resourceConn = nil; resourceActive = false; return end
        if not root or not GameUtil.Data.equipped then return end
        local ct = tick()
        if ct - lastResourceTime < (tonumber(Options.resourcecooldownslider.Value) or 0.1) then return end
        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1
        local targets, all = {}, {}
        for _, r in pairs(workspace.Resources:GetChildren()) do table.insert(all, r) end
        for _, r in pairs(workspace:GetChildren()) do if r:IsA("Model") and r.Name == "Gold Node" then table.insert(all, r) end end
        for _, res in pairs(all) do
            if res:IsA("Model") and res:GetAttribute("EntityID") then
                local pp = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                if pp and (pp.Position - root.Position).Magnitude <= range then
                    table.insert(targets, {eid = res:GetAttribute("EntityID"), dist = (pp.Position - root.Position).Magnitude})
                end
            end
        end
        if #targets > 0 then
            table.sort(targets, function(a,b) return a.dist < b.dist end)
            local sel = {}
            for i = 1, math.min(tc, #targets) do table.insert(sel, targets[i].eid) end
            swingtool(sel)
            lastResourceTime = ct
        end
    end)
end
Options.resourceauratoggle:OnChanged(function(s) if s then startResource() else resourceActive = false; if resourceConn then resourceConn:Disconnect(); resourceConn = nil end end end)

-- Pickup loop
task.spawn(function()
    while true do
        local range = tonumber(Options.pickuprange.Value) or 35
        if Options.autopickuptoggle.Value and root then
            for _, item in ipairs(workspace.Items:GetChildren()) do
                if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then
                    if (item.Position - root.Position).Magnitude <= range then pickup(item:GetAttribute("EntityID")) end
                end
            end
        end
        if Options.chestpickuptoggle.Value and root then
            for _, chest in ipairs(workspace.Deployables:GetChildren()) do
                if chest:IsA("Model") and chest:FindFirstChild("Contents") then
                    for _, item in ipairs(chest.Contents:GetChildren()) do
                        if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then
                            if (chest.PrimaryPart.Position - root.Position).Magnitude <= range then pickup(item:GetAttribute("EntityID")) end
                        end
                    end
                end
            end
        end
        task.wait(0.01)
    end
end)

local fruittoitemid = {Bloodfruit = 94, Bluefruit = 377, Lemon = 99, Coconut = 1, Jelly = 604, Banana = 606, Orange = 602, Oddberry = 32, Berry = 35, Strangefruit = 302, Strawberry = 282, Sunfruit = 128, Pumpkin = 80, ["Prickly Pear"] = 378, Apple = 243, Barley = 247, Cloudberry = 101, Carrot = 147, Blossom = 958}
local function getpbs(range)
    local boxes = {}
    if not root then return boxes end
    for _, d in ipairs(workspace.Deployables:GetChildren()) do
        if d:IsA("Model") and d.Name == "Plant Box" then
            local eid, pp = d:GetAttribute("EntityID"), d.PrimaryPart or d:FindFirstChildWhichIsA("BasePart")
            if eid and pp then
                local dist = (pp.Position - root.Position).Magnitude
                if dist <= range then table.insert(boxes, {entityid = eid, deployable = d, dist = dist}) end
            end
        end
    end
    return boxes
end
local function getbushes(range, fruitname)
    local bushes = {}
    if not root then return bushes end
    for _, m in ipairs(workspace:GetChildren()) do
        if m:IsA("Model") and m.Name:find(fruitname) then
            local pp, eid = m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart"), m:GetAttribute("EntityID")
            if pp and eid and (pp.Position - root.Position).Magnitude <= range then
                table.insert(bushes, {entityid = eid, model = m, dist = (pp.Position - root.Position).Magnitude})
            end
        end
    end
    return bushes
end
local tweening = nil
local function tween(target)
    if not root then return end
    if tweening then tweening:Cancel(); tweening = nil end
    local dur = (root.Position - target.Position).Magnitude / Options.tweenspeed.Value
    if dur < 0.1 then dur = 0.1 end
    local tw = tspmo:Create(root, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = target})
    tw:Play()
    tweening = tw
end

-- Plant box placement
local function placeSingleSnappedBox()
    local char = plr.Character
    if not (char and char:FindFirstChild("HumanoidRootPart")) then return end
    local pos = char.HumanoidRootPart.Position
    local cam = workspace.CurrentCamera
    local cursorPos = cam and cam.CFrame.Position or pos
    local gs = 6.05
    local sp = Vector3.new(math.round(pos.X/gs)*gs, pos.Y, math.round(pos.Z/gs)*gs)
    local rp = RaycastParams.new()
    rp.FilterType, rp.FilterDescendantsInstances, rp.IgnoreWater = Enum.RaycastFilterType.Blacklist, {char}, true
    local rr = workspace:Raycast(Vector3.new(sp.X, sp.Y+100, sp.Z), Vector3.new(0,-200,0), rp)
    local fp = rr and rr.Position or sp
    local rot = Options.rotationdropdown.Value == "45 degrees" and CFrame.Angles(0, math.rad(45), 0) or CFrame.Angles(0,0,0)
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({buildingName = "Plant Box", cframe = CFrame.new(fp)*rot, vec = fp, isMobile = false, cursor = cursorPos})
    end
end
local function placeGridOfBoxes(gridSize)
    local char = plr.Character
    if not (char and char:FindFirstChild("HumanoidRootPart")) then return end
    local torso, cam = char.HumanoidRootPart, workspace.CurrentCamera
    local cursorPos = cam and cam.CFrame.Position or torso.Position
    local rot = Options.rotationdropdown.Value == "45 degrees" and CFrame.Angles(0, math.rad(45), 0) or CFrame.Angles(0,0,0)
    local sp = 6.05
    local startpos = torso.Position - Vector3.new(0,3,0)
    for x = 0, gridSize-1 do
        for z = 0, gridSize-1 do
            task.wait(0.3)
            local gp = Vector3.new(startpos.X + x*sp, startpos.Y, startpos.Z + z*sp)
            local sgp = Vector3.new(math.round(gp.X/sp)*sp, gp.Y, math.round(gp.Z/sp)*sp)
            local rp = RaycastParams.new()
            rp.FilterType, rp.FilterDescendantsInstances, rp.IgnoreWater = Enum.RaycastFilterType.Blacklist, {char}, true
            local rr = workspace:Raycast(Vector3.new(sgp.X, sgp.Y+100, sgp.Z), Vector3.new(0,-200,0), rp)
            local fp = rr and rr.Position or sgp
            if packets.PlaceStructure and packets.PlaceStructure.send then
                packets.PlaceStructure.send({buildingName = "Plant Box", cframe = CFrame.new(fp)*rot, vec = fp, isMobile = false, cursor = cursorPos})
            end
        end
    end
end
local placestructure = function(gs)
    if gs == 1 then placeSingleSnappedBox() else placeGridOfBoxes(gs) end
end

-- Tween toggles
local plantBoxConn, bushConn
Options.tweentoplantbox:OnChanged(function(v)
    if v then
        if plantBoxConn then plantBoxConn:Disconnect() end
        plantBoxConn = runs.Heartbeat:Connect(function()
            if not root then return end
            local boxes = getpbs(Options.tweenrange.Value)
            local closest, minD = nil, math.huge
            for _, b in ipairs(boxes) do
                if not b.deployable:FindFirstChild("Seed") and b.dist < minD then minD, closest = b.dist, b end
            end
            if closest then tween(closest.deployable.PrimaryPart.CFrame + Vector3.new(0,3,0)) end
        end)
    else
        if plantBoxConn then plantBoxConn:Disconnect(); plantBoxConn = nil end
        if tweening then tweening:Cancel(); tweening = nil end
    end
end)
Options.tweentobush:OnChanged(function(v)
    if v then
        if bushConn then bushConn:Disconnect() end
        bushConn = runs.Heartbeat:Connect(function()
            if not root then return end
            local fn = Options.fruitdropdown.Value
            local range = Options.tweenrange.Value
            local bushes = getbushes(range, fn)
            local cb, minD = nil, math.huge
            for _, b in ipairs(bushes) do if b.dist < minD then minD, cb = b.dist, b end end
            if cb then
                tween(cb.model.PrimaryPart.CFrame + Vector3.new(0,3,0))
                if (cb.model.PrimaryPart.Position - root.Position).Magnitude <= range and packets.Pickup then
                    packets.Pickup.send(cb.entityid)
                end
            else
                local boxes = getpbs(range)
                local ce, minD2 = nil, math.huge
                for _, b in ipairs(boxes) do
                    if not b.deployable:FindFirstChild("Seed") and b.dist < minD2 then minD2, ce = b.dist, b end
                end
                if ce then
                    tween(ce.deployable.PrimaryPart.CFrame + Vector3.new(0,3,0))
                    if (ce.deployable.PrimaryPart.Position - root.Position).Magnitude <= range and packets.InteractStructure then
                        pcall(function() packets.InteractStructure.send({entityID = ce.entityid, itemID = fruittoitemid[fn] or 94}) end)
                    end
                end
            end
        end)
    else
        if bushConn then bushConn:Disconnect(); bushConn = nil end
        if tweening then tweening:Cancel(); tweening = nil end
    end
end)

    if not hitbox or not workspace:FindFirstChild("Items") then return end

    for _, item in ipairs(workspace.Items:GetChildren()) do
        local n = item.Name or ""
        if n:find("Ice Cube") or n:find("Bloodfruit") or n:find("Strawberry") or n:find("Prickly Pear") or n:find("Crystal") or n:find("Cloudberry") or n:find("Adurite") or n:find("Fire Hide") or n:find("Berry") or n:find("Iron") or n:find("Lemon") or n:find("Magnetite") then
            pcall(function()
                if item:IsA("BasePart") then item.Position = tp
                elseif item:IsA("Model") and item.PrimaryPart then item:SetPrimaryPartCFrame(CFrame.new(tp)) end
            end)
        end
    end
end
