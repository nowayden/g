local function swingtool(entityIds)

    for _, eid in ipairs(entityIds) do table.insert(arr, {entityID = eid, buffer = 0}) end
    pcall(function()
        packets.SwingTool.send({entityIDs = arr, cframe = root and root.CFrame or CFrame.new(), timestamp = clock.getServerTime(true)})
    end)
end

-- ==================== RE-ENGINEERED COMPACT CACHING REQUISITIONS ====================

local function getpbs(range)
    local boxes = {}
    if not root then return boxes end
    local currentPos = root.Position
    local deployables = workspace:FindFirstChild("Deployables")
    if not deployables then return boxes end
    
    local children = deployables:GetChildren()
    for i = 1, #children do
        local d = children[i]
        if d.Name == "Plant Box" then
            local eid = d:GetAttribute("EntityID")
            local pp = d.PrimaryPart or d:FindFirstChildWhichIsA("BasePart")
            if eid and pp then
                local dist = (pp.Position - currentPos).Magnitude
                if dist <= range then 
                    table.insert(boxes, {entityid = eid, deployable = d, dist = dist, cframe = pp.CFrame}) 
                end
            end
        end
    end
    return boxes
end

local function getbushes(range, fruitname)
    local bushes = {}
    if not root then return bushes end
    local currentPos = root.Position
    local children = workspace:GetChildren()
    
    for i = 1, #children do
        local m = children[i]
        if m:IsA("Model") and m.Name:find(fruitname) then
            local pp = m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")
            local eid = m:GetAttribute("EntityID")
            if pp and eid then
                local dist = (pp.Position - currentPos).Magnitude
                if dist <= range then
                    table.insert(bushes, {entityid = eid, model = m, dist = dist, cframe = pp.CFrame, position = pp.Position})
                end
            end
        end
    end
    return bushes
end

-- Resource Aura
local resourceActive, resourceConn, lastResourceTime = false, nil, 0
local function startResource()
    if resourceActive then return end
    resourceActive = true
    resourceConn = runs.Heartbeat:Connect(function()
        if not Options.resourceauratoggle.Value then resourceConn:Disconnect(); resourceConn = nil; resourceActive = false; return end
        if not root or not GameUtil.Data.equipped then return end
        local ct = tick()
        if ct - lastResourceTime < (tonumber(Options.resourcecooldownslider.Value) or 0.1) then return end
        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1
        local targets, all = {}, {}
        for _, r in pairs(workspace.Resources:GetChildren()) do table.insert(all, r) end
        for _, r in pairs(workspace:GetChildren()) do if r:IsA("Model") and r.Name == "Gold Node" then table.insert(all, r) end end
        for _, res in pairs(all) do
            if res:IsA("Model") and res:GetAttribute("EntityID") then
                local pp = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                if pp and (pp.Position - root.Position).Magnitude <= range then
                    table.insert(targets, {eid = res:GetAttribute("EntityID"), dist = (pp.Position - root.Position).Magnitude})
                end
            end
        end
        if #targets > 0 then
            table.sort(targets, function(a,b) return a.dist < b.dist end)
            local sel = {}
            for i = 1, math.min(tc, #targets) do table.insert(sel, targets[i].eid) end
            swingtool(sel)
            lastResourceTime = ct
        end
    end)
end
Options.resourceauratoggle:OnChanged(function(s) if s then startResource() else resourceActive = false; if resourceConn then resourceConn:Disconnect(); resourceConn = nil end end end)

task.spawn(function()
    while true do
        local range = tonumber(Options.pickuprange.Value) or 35
        if Options.autopickuptoggle.Value and root then
            local items = workspace.Items:GetChildren()
            for i = 1, #items do
                local item = items[i]
                if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then
                    if (item.Position - root.Position).Magnitude <= range then pickup(item:GetAttribute("EntityID")) end
                end
            end
        end
        if Options.chestpickuptoggle.Value and root then
            local deployables = workspace.Deployables:GetChildren()
            for i = 1, #deployables do
                local chest = deployables[i]
                if chest:IsA("Model") and chest:FindFirstChild("Contents") and chest.PrimaryPart then
                    if (chest.PrimaryPart.Position - root.Position).Magnitude <= range then
                        local contents = chest.Contents:GetChildren()
                        for j = 1, #contents do
                            local item = contents[j]
                            if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then
                                pickup(item:GetAttribute("EntityID"))
                            end
                        end
                    end
                end
            end
        end
        task.wait(0.1)
    end
end)

-- ==================== STABLE HIGH-PERFORMANCE FARMING ENGINE ====================

local fruittoitemid = {Bloodfruit = 94, Bluefruit = 377, Lemon = 99, Coconut = 1, Jelly = 604, Banana = 606, Orange = 602, Oddberry = 32, Berry = 35, Strangefruit = 302, Strawberry = 282, Sunfruit = 128, Pumpkin = 80, ["Prickly Pear"] = 378, Apple = 243, Barley = 247, Cloudberry = 101, Carrot = 147, Blossom = 958}
local tweening = nil

local function tween(targetCFrame)
    if not root then return end
    if tweening then tweening:Cancel() end
    
    local speed = Options.tweenspeed.Value or 50
    local dur = (root.Position - targetCFrame.Position).Magnitude / speed
    if dur < 0.05 then dur = 0.05 end
    
    tweening = tspmo:Create(root, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = targetCFrame})
    tweening:Play()
    
    -- Wait until the animation completes or gets canceled externally
    local completedSignal
    completedSignal = tweening.Completed:Connect(function()
        completedSignal:Disconnect()
        tweening = nil
    end)
end

function placeSingleSnappedBox()
    local char = plr.Character
    if not (char and char:FindFirstChild("HumanoidRootPart")) then return end
    local pos = char.HumanoidRootPart.Position
    local cam = workspace.CurrentCamera
    local cursorPos = cam and cam.CFrame.Position or pos
    
    local gs = 5.05
    local sp = Vector3.new(math.round(pos.X/gs)*gs, pos.Y, math.round(pos.Z/gs)*gs)
    
    local rp = RaycastParams.new()
    rp.FilterType, rp.FilterDescendantsInstances, rp.IgnoreWater = Enum.RaycastFilterType.Blacklist, {char}, true
    local rr = workspace:Raycast(Vector3.new(sp.X, sp.Y+100, sp.Z), Vector3.new(0,-200,0), rp)
    local fp = rr and rr.Position or sp
    local rot = CFrame.Angles(0, 0, 0)
    
    if packets.PlaceStructure and packets.PlaceStructure.send then
        packets.PlaceStructure.send({buildingName = "Plant Box", cframe = CFrame.new(fp)*rot, vec = fp, isMobile = false, cursor = cursorPos})
    end
end

-- Multi-threaded 24/7 continuous worker for movement logic
task.spawn(function()
    while task.wait(0.05) do
        if root then
            local range = Options.tweenrange.Value or 250
            local selectedFruit = Options.fruitdropdown.Value or "Bloodfruit"
            
            -- VERSION 1: Tween to Plant Box 24/7 Non-Stop
            if Options.tweentoplantbox.Value and not Options.tweentobush.Value then
                local boxes = getpbs(range)
                local closestBox, minBoxDist = nil, math.huge
                
                for i = 1, #boxes do
                    local b = boxes[i]
                    if not b.deployable:FindFirstChild("Seed") and b.dist < minBoxDist then
                        minBoxDist = b.dist
                        closestBox = b
                    end
                end
                
                if closestBox then
                    tween(closestBox.cframe + Vector3.new(0, 3, 0))
                end
                
            -- VERSION 2: Tween to Bush + Plant Box Mixed Mode
            elseif Options.tweentobush.Value then
                local bushes = getbushes(range, selectedFruit)
                local closestBush, minBushDist = nil, math.huge
                
                for i = 1, #bushes do
                    local b = bushes[i]
                    if b.dist < minBushDist then
                        minBushDist = b.dist
                        closestBush = b
                    end
                end
                
                if closestBush then
                    tween(closestBush.cframe + Vector3.new(0, 3, 0))
                    if closestBush.dist <= 15 and packets.Pickup then
                        pickup(closestBush.entityid)
                    end
                else
                    -- No bushes found, proceed to process empty boxes inside the same thread seamlessly
                    local boxes = getpbs(range)
                    local closestBox, minBoxDist = nil, math.huge
                    
                    for i = 1, #boxes do
                        local b = boxes[i]
                        if not b.deployable:FindFirstChild("Seed") and b.dist < minBoxDist then
                            minBoxDist = b.dist
                            closestBox = b
                        end
                    end
                    
                    if closestBox then
                        tween(closestBox.cframe + Vector3.new(0, 3, 0))
                        if closestBox.dist <= 15 and packets.InteractStructure then
                            local itemID = fruittoitemid[selectedFruit] or 94
                            pcall(function() 
                                packets.InteractStructure.send({entityID = closestBox.entityid, itemID = itemID}) 
                            end)
                        end
                    end
                end
            end
        end
    end
end)

-- FPS Lag Fix for Auto Plant Routine (Optimized query cycle)
local isPlantingRunning = false
local lastPlantCycle = 0

local function FastPlantLoop()
    while isPlantingRunning do
        local ct = tick()
        local executionDelay = tonumber(Options.plantdelay.Value) or 0.1
        
        if ct - lastPlantCycle >= executionDelay then
            if Options.planttoggle.Value and root and packets.InteractStructure then
                local currentRange = tonumber(Options.plantrange.Value) or 30
                local selectedFruit = Options.fruitdropdown.Value or "Bloodfruit"
                local itemID = fruittoitemid[selectedFruit] or 94
                
                local nearbyBoxes = getpbs(currentRange)
                if #nearbyBoxes > 0 then
                    for i = 1, #nearbyBoxes do
                        if not isPlantingRunning then break end
                        local box = nearbyBoxes[i]
                        if not box.deployable:FindFirstChild("Seed") then
                            pcall(function()
                                packets.InteractStructure.send({entityID = box.entityid, itemID = itemID})
                            end)
                        end
                    end
                end
            end
            lastPlantCycle = ct
        end
        task.wait(0.02) -- Fixed yield to keep thread running smoothly without locking execution frames
    end
end

Options.planttoggle:OnChanged(function(v)
    isPlantingRunning = v
    if v then task.spawn(FastPlantLoop) end
end)

    local cauldrons = {}
    for _, obj in ipairs(dep:GetChildren()) do
        if obj.Name == "Cauldron" and obj:FindFirstChild("Hitbox") and (obj.Hitbox.Position - rp).Magnitude < 30 then
            table.insert(cauldrons, {cauldron = obj, hitbox = obj.Hitbox})
        end
    end
    return cauldrons
end

local function teleportToHitbox(hitbox)
    if not hitbox or not workspace:FindFirstChild("Items") then return end
    local tp = hitbox.Position + Vector3.new(math.random(-2,2)*0.3, 0.8, math.random(-2,2)*0.3)
    for _, item in ipairs(workspace.Items:GetChildren()) do
        local n = item.Name or ""
        if n:find("Ice Cube") or n:find("Bloodfruit") or n:find("Strawberry") or n:find("Prickly Pear") or n:find("Crystal") or n:find("Cloudberry") or n:find("Adurite") or n:find("Fire Hide") or n:find("Berry") or n:find("Iron") or n:find("Lemon") or n:find("Magnetite") then
            pcall(function()
                if item:IsA("BasePart") then item.Position = tp
                elseif item:IsA("Model") and item.PrimaryPart then item:SetPrimaryPartCFrame(CFrame.new(tp)) end
            end)
        end
    end
end

local function brewPotions()
    if isBrewing then return end
    isBrewing = true
    local cauldrons = findAllCauldrons()
    if #cauldrons == 0 then isBrewing = false; return end
    local recipe = POTIONS[selectedPotion]
    if not recipe then isBrewing = false; return end
    for _, ing in ipairs(recipe) do
        local slot = getInventorySlot(ing.name)
        if slot then
            for _, c in ipairs(cauldrons) do dropFromSlot(slot, ing.amount); task.wait(0.5); teleportToHitbox(c.hitbox); task.wait(0.4) end
        end
        task.wait(0.8)
    end
    task.wait(2)
    for _, c in ipairs(cauldrons) do
        local eid = c.cauldron:GetAttribute("EntityID")
        if eid then pcall(function() packets.SweepPotion.send(eid) end) end
    end
    isBrewing = false
end

-- ==================== AUTO REBIRTH SYSTEM ====================

local autoRebirthRunning = false
local hasConfirmedRebirth = false

local function pressBedButton()
    local spawnGui = plr.PlayerGui:FindFirstChild("SpawnGui")
    if spawnGui and spawnGui:FindFirstChild("Customization") then
        local bedButton = spawnGui.Customization:FindFirstChild("BedButton")
        if bedButton then
            pcall(function() bedButton.Activated:Fire() end)
            pcall(function() firesignal(bedButton.Activated) end)
            print("BedButton pressed directly!")
            return
        end
    end

    for _, gui in ipairs(plr.PlayerGui:GetDescendants()) do
        if gui.Name == "BedButton" then
            pcall(function() gui.Activated:Fire() end)
            pcall(function() firesignal(gui.Activated) end)
        end
    end

    pcall(function()
        rs.Events.SpawnFirst:InvokeServer(true)
    end)
end

local function getPlayerLevel()
    local success, GameUtilLocal = pcall(function() return require(rs.Modules.GameUtil) end)
    if success and GameUtilLocal.Data and GameUtilLocal.Data.level then
        return GameUtilLocal.Data.level
    end
    return plr:GetAttribute("Level") or 0
end

local function doRebirth()
    print("🎉 Level 100 reached → Triggering Rebirth")
    local success, packetsLocal = pcall(function() return require(rs.Modules.Packets) end)
    if success and packetsLocal.Rebirth then
        pcall(function() packetsLocal.Rebirth.send() end)
    end
    task.wait(2.5)
    pressBedButton()
end

local function AutoRebirthLoop()
    while autoRebirthRunning do
        task.wait(0.5)
        local spawnGui = plr.PlayerGui:FindFirstChild("SpawnGui")
        if spawnGui and spawnGui.Enabled then
            pressBedButton()
        end
        local currentLevel = getPlayerLevel()
        if currentLevel >= 100 then
            print("LEVEL 100 DETECTED!")
            doRebirth()
            task.wait(6)
        end
    end
end

local function StartAutoRebirth()
    if autoRebirthRunning then return end
    autoRebirthRunning = true
    print("✅ AUTO REBIRTH ENABLED")
    task.spawn(AutoRebirthLoop)
end

local function StopAutoRebirth()
    autoRebirthRunning = false
    print("⛔ AUTO REBIRTH DISABLED")
end

Tabs.Event:AddSection("Auto Rebirth")
Tabs.Event:AddToggle("AutoRebirthToggle", {

    Default = false,
    Callback = function(state)

            if not hasConfirmedRebirth then
                Window:Dialog({
                    Title = "Confirmation Required",
                    Content = "Are you sure you want to toggle Auto Rebirth?",
                    Buttons = {
                        {
                            Title = "Confirm",
                            Callback = function()
                                hasConfirmedRebirth = true
                                StartAutoRebirth()
                            end
                        },
                        {
                            Title = "Cancel",
                            Callback = function()
                                Options.AutoRebirthToggle:SetValue(false)
                            end
                        }
                    }
                })
            else
                StartAutoRebirth()
            end
        else
            StopAutoRebirth()
        end
    end
end)
