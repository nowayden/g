local function pickupencode(entityId)
    local packetId = packets.Pickup and (packets.Pickup._id or 0xE7) or 0xE7
    local header = string.char(0x00, packetId)
    local data = string.pack("<I4", entityId)
    return header .. data
end

local function silentPickup(entityId)
    if not entityId then return end
    pcall(function()
        local packet = pickupencode(entityId)
        rs:WaitForChild("ByteNetReliable"):FireServer(buffer.fromstring(packet))
    end)
end

-- RAW SWINGTOOL
local function float16(v)
    if v == 0 then return 0 end
    local sign = v < 0 and 0x8000 or 0
    v = math.abs(v)
    local exp = math.floor(math.log(v, 2))
    local mant = math.floor((v / 2 ^ exp) * 1024)
    return sign + (exp + 15) * 32 + mant
end

local function swingencode(ids, cf, ts)
    if typeof(ids) ~= "table" then ids = {ids} end
    local parts = {string.char(0x00, 0x13)}
    table.insert(parts, string.pack("<H", #ids))
    for _, v in ipairs(ids) do table.insert(parts, string.pack("<I4x", v)) end
    local x, y, z = cf:ToEulerAnglesXYZ()
    table.insert(parts, string.pack("<fffHHHd", cf.X, cf.Y, cf.Z, float16(x), float16(y), float16(z), ts))
    return table.concat(parts)
end

local function swingtool(entityIds)
    local rootPart = root or (plr.Character and plr.Character:FindFirstChild("HumanoidRootPart"))
    if not rootPart then return end
    local ts = workspace:GetServerTimeNow()
    local packet = swingencode(entityIds, rootPart.CFrame, ts)
    pcall(function() rs:WaitForChild("ByteNetReliable"):FireServer(buffer.fromstring(packet)) end)
end

local selecteditems = {}
Options.itemdropdown:OnChanged(function(v)
    selecteditems = {}
    for item, state in pairs(v) do if state then table.insert(selecteditems, item) end end
end)

-- FIXED AUTO PICKUP LOOP
task.spawn(function()
    while true do
        if not autopickuptoggle.Value and not chestpickuptoggle.Value then task.wait(0.1) continue end

        local range = tonumber(Options.pickuprange.Value) or 18
        local cooldown = tonumber(Options.pickupcooldown.Value) or 0.05

        if autopickuptoggle.Value and root then
            for _, item in ipairs(workspace.Items:GetChildren()) do
                if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then
                    if (item.Position - root.Position).Magnitude <= range then
                        silentPickup(item:GetAttribute("EntityID"))
                        task.wait(cooldown)
                    end
                end
            end
        end

        if chestpickuptoggle.Value and root then
            local dep = workspace:FindFirstChild("Deployables")
            if dep then
                for _, chest in ipairs(dep:GetChildren()) do
                    if chest.Name == "Chest" and chest.PrimaryPart and chest:FindFirstChild("Contents") then
                        if (chest.PrimaryPart.Position - root.Position).Magnitude <= range then
                            for _, item in ipairs(chest.Contents:GetChildren()) do
                                if (item:IsA("BasePart") or item:IsA("MeshPart")) and item:GetAttribute("EntityID") and table.find(selecteditems, item.Name) then
                                    silentPickup(item:GetAttribute("EntityID"))
                                    task.wait(cooldown)
                                end
                            end
                        end
                    end
                end
            end
        end
        task.wait(0.01)
    end
end)

-- Drop
local debounce = 0
local function drop(itemname)
    local list = plr.PlayerGui:FindFirstChild("MainGui") and plr.PlayerGui.MainGui.RightPanel.Inventory.List
    if not list then return end
    for _, child in ipairs(list:GetChildren()) do
        if child:IsA("ImageLabel") and child.Name == itemname then
            SecureNetwork.DropBagItem(child.LayoutOrder)
        end
    end
end

runs.Heartbeat:Connect(function()
    if Options.droptoggle.Value and tick() - debounce >= 0.2 then
        drop(Options.dropdropdown.Value)
        debounce = tick()
    end
end)

runs.Heartbeat:Connect(function()
    if Options.droptogglemanual.Value and tick() - debounce >= 0.2 then
        drop(Options.droptextbox.Value)
        debounce = tick()
    end
end)
