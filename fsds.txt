local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- FIXED Packet Interception
local originalPickupSend = packets.Pickup and packets.Pickup.send
if originalPickupSend then
    packets.Pickup.send = function(entityID, ...)
        if Options.harvesttoggle and Options.harvesttoggle.Value then
            local targetBush = nil
            for _, v in ipairs(workspace:GetChildren()) do
                if v:GetAttribute("EntityID") == entityID then targetBush = v; break end
            end
            
            if targetBush then
                local addedValue = 3
                -- Clean logic, no ternary operator risk
                if targetBush.Name:lower():find("small") then
                    addedValue = 2
                end
                
                if not farmStats.timerStarted then
                    farmStats.timerStarted = true
                    farmStats.startTime = tick()
                end
                farmStats.harvestedCount = farmStats.harvestedCount + addedValue
            end
        end
        return originalPickupSend(entityID, ...)
    end
end

-- UI Loop
task.spawn(function()
    while true do
        task.wait(0.5)
        local runElapsed = math.max(tick() - farmStats.startTime, 1)
        local itemsPerMin = math.round((farmStats.harvestedCount / runElapsed) * 60)
        local itemsPerHour = math.round((farmStats.harvestedCount / runElapsed) * 3600)
        
        if HarvestedStats then HarvestedStats:SetTitle(string.format("Harvested (Session): %d (Per Min: %d | Per Hour: %d)", farmStats.harvestedCount, itemsPerMin, itemsPerHour)) end
    end
end)

-- Harvesting Loop

                    local eid = v:GetAttribute("EntityID")
                    if eid then originalPickupSend(eid) end
                end
            end
        end
    end
end)
