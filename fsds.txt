-- Helper function to format seconds into HH:MM:SS
local function formatTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = math.floor(seconds % 60)
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

-- Robust inventory scanner that checks background data modules and UI layouts
local function getInventoryCount(itemName)
    if GameUtil and GameUtil.Data then
        local inv = GameUtil.Data.inventory or GameUtil.Data.Inventory or GameUtil.Data.bag or GameUtil.Data.Bag
        if type(inv) == "table" then
            if inv[itemName] then return tonumber(inv[itemName]) or 0 end
            for _, v in pairs(inv) do
                if type(v) == "table" and (v.name == itemName or v.Name == itemName) then
                    return tonumber(v.quantity or v.amount or v.Amount or v.count or v.Count or 1) or 0
                end
            end
        end
    end
    
    local list = plr.PlayerGui:FindFirstChild("MainGui") 
        and plr.PlayerGui.MainGui:FindFirstChild("RightPanel") 
        and plr.PlayerGui:FindFirstChild("MainGui").RightPanel:FindFirstChild("Inventory") 
        and plr.PlayerGui.MainGui.RightPanel.Inventory:FindFirstChild("List")
    if list then
        local total = 0
        for _, slot in ipairs(list:GetChildren()) do
            if slot:IsA("GuiObject") and (slot.Name == itemName or slot.Name:find(itemName)) then
                local quantityImg = slot:FindFirstChild("QuantityImage")
                local textObj = (quantityImg and quantityImg:FindFirstChild("QuantityText")) or slot:FindFirstChild("Amount") or slot:FindFirstChild("Count") or slot:FindFirstChildWhichIsA("TextLabel")
                if textObj then
                    total = total + (tonumber(textObj.Text:match("%d+")) or 1)
                else
                    total = total + 1
                end
            end
        end
        if total > 0 then return total end
    end
    return 0
end

-- Resource Aura (unchanged)
local resourceActive, resourceConn, lastResourceTime = false, nil, 0
local function startResource()
    if resourceActive then return end
    resourceActive = true
    resourceConn = runs.Heartbeat:Connect(function()
        if not Options.resourceauratoggle.Value then resourceConn:Disconnect(); resourceConn = nil; resourceActive = false; return end
        if not root or not GameUtil.Data.equipped then return end
        local ct = tick()
        if ct - lastResourceTime < (tonumber(Options.resourcecooldownslider.Value) or 0.1) then return end
        local range = tonumber(Options.resourceaurarange.Value) or 20
        local tc = tonumber(Options.resourcetargetdropdown.Value) or 1
        local targets, all = {}, {}
        
        if workspace:FindFirstChild("Resources") then
            for _, r in pairs(workspace.Resources:GetDescendants()) do 
                if r:IsA("Model") then 
                    table.insert(all, r) 
                end 
            end
        end
 
        local resourceKeywords = {"Node", "Stone", "Lode", "Rock", "Coal", "Guardian"}
        for _, r in pairs(workspace:GetChildren()) do 
            if r:IsA("Model") then
                local matchFound = false
                for _, kw in ipairs(resourceKeywords) do
                    if string.find(r.Name, kw) then
                        matchFound = true
                        break
                    end
                end
                if matchFound then
                    table.insert(all, r)
                end
            end 
        end
        
        for _, res in pairs(all) do
            if res:IsA("Model") and res:GetAttribute("EntityID") then
                local pp = res.PrimaryPart or res:FindFirstChildWhichIsA("BasePart")
                if pp and (pp.Position - root.Position).Magnitude <= range then
                    table.insert(targets, {eid = res:GetAttribute("EntityID"), dist = (pp.Position - root.Position).Magnitude})
                end
            end
        end
        if #targets > 0 then
            table.sort(targets, function(a,b) return a.dist < b.dist end)
            local sel = {}
            for i = 1, math.min(tc, #targets) do table.insert(sel, targets[i].eid) end
            swingtool(sel)
            lastResourceTime = ct
        end
    end)
end
Options.resourceauratoggle:OnChanged(function(s) if s then startResource() else resourceActive = false; if resourceConn then resourceConn:Disconnect(); resourceConn = nil end end end)

-- ========= [ DEPLOYABLE AURA FROM V9 ] =========
local RS = runs
local isTraderActive = false

local deployableAuraActive = false
local deployableAuraConnection = nil
local lastDeployableSwingTime = 0

function StartDeployableAura(range)
    if deployableAuraActive then return end
    deployableAuraActive = true
    range = range or 25

    deployableAuraConnection = RS.Heartbeat:Connect(function()
        if not deployableAuraActive or isTraderActive then return end

        local currentTime = tick()
        if currentTime - lastDeployableSwingTime < 0.1 then return end

        local char = plr.Character
        local rootPart = char and char:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end

        local targetList = {}
        local folders = {"Deployables", "Rubble", "dropChests"}
      
        for _, folderName in ipairs(folders) do
            local folder = workspace:FindFirstChild(folderName)
            if folder then
                for _, child in pairs(folder:GetChildren()) do
                    local eid = child:GetAttribute("EntityID")
            
                    if eid then
                        local pp = child.PrimaryPart or child:FindFirstChildWhichIsA("BasePart")
                        if pp and (pp.Position - rootPart.Position).Magnitude <= range then
                            table.insert(targetList, {entityID = eid, buffer = 0})
                        end
                    end
                end
            end
        end

        if #targetList > 0 then
            pcall(function()
                packets.SwingTool.send({
                    entityIDs = targetList,
                    cframe = rootPart.CFrame,
                    timestamp = clock.getServerTime(true)
                })
            end)
            lastDeployableSwingTime = currentTime
        end
    end)
end

function StopDeployableAura()
    deployableAuraActive = false
    if deployableAuraConnection then deployableAuraConnection:Disconnect() end
end

local plantedboxes = {}
local fruittoitemid = {Bloodfruit = 94, Bluefruit = 377, Lemon = 99, Coconut = 1, Jelly = 604, Banana = 606, Orange = 602, Oddberry = 32, Berry = 35, Strangefruit = 302, Strawberry = 282, Sunfruit = 128, Pumpkin = 80, ["Prickly Pear"] = 378, Apple = 243, Barley = 247, Cloudberry = 101, Carrot = 147, Blossom = 958}

-- ======= OPTIMIZED: Plant Box / Bush Caches and Fast Queries =======
local plantCache = {}            -- { id = entityid, box = model, part = part, lastSeen = tick() }
local bushCache = {}             -- { id = entityid, model = model, part = part, lastSeen = tick(), name = model.Name }
local plantCacheUpdate = 0
local bushCacheUpdate = 0
local PLANT_CACHE_TTL = 1.0      -- seconds between full refreshes
local BUSH_CACHE_TTL = 0.8

local function refreshPlantCache()
    local now = tick()
    if now - plantCacheUpdate < PLANT_CACHE_TTL then return end
    plantCache = {}
    local dep = workspace:FindFirstChild("Deployables")
    if not dep then plantCacheUpdate = now; return end
    local children = dep:GetChildren()
    for i = 1, #children do
        local v = children[i]
        if v.Name == "Plant Box" and v:IsA("Model") then
            local ppart = v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart")
            local entityid = v:GetAttribute("EntityID")
            if ppart and entityid then
                plantCache[#plantCache + 1] = { id = entityid, box = v, part = ppart, lastSeen = now }
            end
        end
    end
    plantCacheUpdate = now
end

local function refreshBushCache(fruitname)
    local now = tick()
    if now - bushCacheUpdate < BUSH_CACHE_TTL then return end
    bushCache = {}
    local children = workspace:GetChildren()
    for i = 1, #children do
        local model = children[i]
        if model:IsA("Model") and model.Name:find(fruitname) then
            local ppart = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            local entityid = model:GetAttribute("EntityID")
            if ppart and entityid then
                bushCache[#bushCache + 1] = { id = entityid, model = model, part = ppart, lastSeen = now, name = model.Name }
            end
        end
    end
    bushCacheUpdate = now
end

-- ======= OPTIMIZED: Planting loop (fast, prioritized, batched) =======
local plantQueue = {}  -- small FIFO to avoid spamming same box
local plantHeartbeatConn = nil
local PLANT_BATCH_DELAY = 0.06  -- small delay between plant sends to remain fast but stable

local function enqueuePlant(entityid, itemID)
    plantQueue[#plantQueue + 1] = { id = entityid, item = itemID }
end

local function plantWorker()
    if plantHeartbeatConn then plantHeartbeatConn:Disconnect(); plantHeartbeatConn = nil end
    plantHeartbeatConn = runs.Heartbeat:Connect(function(dt)
        if not Options.planttoggle.Value then return end
        if not root then return end

        -- refresh caches
        refreshPlantCache()
        local range = tonumber(Options.plantrange.Value) or 30
        local itemID = fruittoitemid[Options.fruitdropdown.Value] or 94

        -- Fill queue with nearest empty boxes (fast scan of plantCache)
        local myPos = root.Position
        for i = 1, #plantCache do
            local data = plantCache[i]
            if data and data.box and data.box.Parent and (data.part.Position - myPos).Magnitude <= range then
                if not data.box:FindFirstChild("Seed") and not plantedboxes[data.id] then
                    -- mark plantedboxes immediately to avoid duplicates
                    plantedboxes[data.id] = true
                    enqueuePlant(data.id, itemID)
                end
            end
        end

        -- Process a small batch from queue
        if #plantQueue > 0 then
            local entry = table.remove(plantQueue, 1)
            if entry and entry.id and packets.InteractStructure and packets.InteractStructure.send then
                pcall(function()

-- ======= OPTIMIZED: Walk to Plant Box / Walk to Bush + Plant Box (both versions) =======
-- Uses Humanoid:MoveTo with MoveToFinished and robust stuck detection.
local moveState = {
    moving = false,
    targetPos = nil,
    lastPos = nil,
    stuckTimer = 0,
    moveConn = nil,
    finishedConn = nil
}

local function stopMove()
    moveState.moving = false
    moveState.targetPos = nil
    moveState.stuckTimer = 0
    if moveState.moveConn then moveState.moveConn:Disconnect(); moveState.moveConn = nil end
    if moveState.finishedConn then moveState.finishedConn:Disconnect(); moveState.finishedConn = nil end
end

local function startMoveTo(pos)
    if not hum or not root or not pos then return end
    stopMove()
    moveState.moving = true
    moveState.targetPos = pos
    moveState.lastPos = root.Position
    moveState.stuckTimer = 0

    local moveStart = tick()
    hum:MoveTo(pos)

    moveState.finishedConn = hum.MoveToFinished:Connect(function(reached)
        -- If reached, clear state
        if reached then
            stopMove()
        else
            -- If not reached, attempt small jump and retry
            if hum then hum.Jump = true end
            -- retry MoveTo once after a short delay
            task.delay(0.25, function()
                if moveState.moving and hum and root then
                    hum:MoveTo(moveState.targetPos)
                end
            end)
        end
    end)

    -- continuous stuck detection
    moveState.moveConn = runs.Heartbeat:Connect(function(dt)
        if not moveState.moving or not root then return end
        local now = tick()
        local distToTarget = (root.Position - moveState.targetPos).Magnitude
        -- if close enough, stop
        if distToTarget <= 3 then stopMove(); return end

        -- detect if position hasn't changed meaningfully
        local moved = (root.Position - moveState.lastPos).Magnitude
        if moved < (1 * dt) then
            moveState.stuckTimer = moveState.stuckTimer + dt
            if moveState.stuckTimer >= 1.2 then
                -- try jump and small random sidestep
                if hum then hum.Jump = true end
                local sidestep = Vector3.new((math.random()-0.5)*2, 0, (math.random()-0.5)*2)
                local fallback = moveState.targetPos + sidestep
                hum:MoveTo(fallback)
                moveState.stuckTimer = 0
            end
        else
            moveState.stuckTimer = 0
        end
        moveState.lastPos = root.Position
    end)
end

-- Path helpers using caches (fast)
local function pathToPlantbox(range)
    refreshPlantCache()
    if not root then return end
    local myPos = root.Position
    local nearest = nil
    for i = 1, #plantCache do
        local data = plantCache[i]
        if data and data.box and data.box.Parent and not data.box:FindFirstChild("Seed") then
            local dist = (data.part.Position - myPos).Magnitude
            if dist <= range and (not nearest or dist < nearest.dist) then
                nearest = { pos = data.part.Position, dist = dist }
            end
        end
    end
    if nearest then startMoveTo(nearest.pos) end
end

local function pathToBushesOrBoxes(range, fruitname)
    refreshBushCache(fruitname)
    if not root then return end
    local myPos = root.Position
    local nearestBush = nil
    for i = 1, #bushCache do
        local b = bushCache[i]
        if b and b.part and b.model and b.model.Parent then
            local dist = (b.part.Position - myPos).Magnitude
            if dist <= range and (not nearestBush or dist < nearestBush.dist) then
                nearestBush = { pos = b.part.Position, dist = dist }
            end
        end
    end
    if nearestBush then
        startMoveTo(nearestBush.pos)
        return
    end
    -- fallback to plant boxes
    pathToPlantbox(range)
end

                    -- Priority 2: Soul Minion
                    elseif c.Name == "Soul Minion" and not traderInstance and not queenInstance then

                     elseif c.Name == "Void Ant" and not traderInstance and not queenInstance and not minionInstance then

                        end

     end
    
    if traderInstance then return traderInstance end
