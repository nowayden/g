-- UNDETECTED PICKUP FUNCTION (BYPASSES WEBHOOK)
local pickupPacketID = 92

local function undetectedPickup(entityId)
    if not entityId then return end
    
    task.spawn(function()
        pcall(function()
            local reliableEvent = rs:FindFirstChild("ByteNetReliable")
            if reliableEvent then
                local bufferData = string.char(pickupPacketID) .. string.pack("<I4", entityId)
                reliableEvent:FireServer(buffer.fromstring(bufferData))
            end
        end)
    end)
end

-- Override the original pickup
if packets.Pickup and packets.Pickup.send then
    packets.Pickup.send = undetectedPickup
end

-- SWINGTOOL BYPASS (WORKING)
local function float16(v)
    if v == 0 then return 0 end
    local sign = v < 0 and 0x8000 or 0
    v = math.abs(v)
    local exp = math.floor(math.log(v, 2))
    local mant = math.floor((v / 2 ^ exp) * 1024)
    return sign + (exp + 15) * 32 + mant
end

local function swingencode(ids, cf, ts)
    if type(ids) ~= "table" then ids = {ids} end
    local parts = {string.char(0x00, 0x13)}
    table.insert(parts, string.pack("<H", #ids))
    for i = 1, #ids do
        table.insert(parts, string.pack("<I4x", ids[i]))
    end
    local x, y, z = cf:ToEulerAnglesXYZ()
    table.insert(parts, string.pack("<fffHHHd", cf.X, cf.Y, cf.Z, float16(x), float16(y), float16(z), ts))
    return table.concat(parts)
end

local function swingtool(entityIds)
    local char = plr.Character
    local rootPart = char and char:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end
    
    pcall(function()
        local ts = workspace:GetServerTimeNow()
        local packet = swingencode(entityIds, rootPart.CFrame, ts)
        rs:WaitForChild("ByteNetReliable"):FireServer(buffer.fromstring(packet))
    end)
end

local resourceActive = false
local resourceConn = nil
local lastResourceTime = 0

Options.resourceauratoggle:OnChanged(function(s)
    if s then
        if resourceActive then return end
        resourceActive = true
        resourceConn = runs.Heartbeat:Connect(function()
            if not Options.resourceauratoggle.Value then
                resourceActive = false
                if resourceConn then resourceConn:Disconnect() end
                return
            end
            if not root or not GameUtil.Data.equipped then return end
            
            local ct = tick()
            local cooldown = tonumber(Options.resourcecooldownslider.Value) or 0.1
            if ct - lastResourceTime < cooldown then return end
            
            local range = tonumber(Options.resourceaurarange.Value) or 20
            local tc = tonumber(Options.resourcetargetdropdown.Value) or 1
            local targets = {}
            
            local resourcesFolder = workspace:FindFirstChild("Resources")
            if resourcesFolder then
                for _, r in pairs(resourcesFolder:GetDescendants()) do
                    if r:IsA("Model") and r:GetAttribute("EntityID") then
                        local pp = r.PrimaryPart or r:FindFirstChildWhichIsA("BasePart")
                        if pp and (pp.Position - root.Position).Magnitude <= range then
                            table.insert(targets, {eid = r:GetAttribute("EntityID"), dist = (pp.Position - root.Position).Magnitude})
                        end
                    end
                end
            end
            
            if #targets > 0 then
                table.sort(targets, function(a,b) return a.dist < b.dist end)
                local sel = {}
                for i = 1, math.min(tc, #targets) do
                    table.insert(sel, targets[i].eid)
                end
                swingtool(sel)
                lastResourceTime = ct
            end
        end)
    else
        resourceActive = false
        if resourceConn then resourceConn:Disconnect() end
    end
end)

local function StartDeployableAura()
    if deployableActive then return end
    deployableActive = true
    deployableConn = runs.Heartbeat:Connect(function()
        if not deployableActive then return end
        local ct = tick()
        if ct - lastDeployableTime < 0.1 then return end

        local rootPart = char and char:FindFirstChild("HumanoidRootPart")

        local targetList = {}
        local folders = {"Deployables", "Rubble", "dropChests"}
        for _, folderName in ipairs(folders) do
            local folder = workspace:FindFirstChild(folderName)
            if folder then
                for _, child in pairs(folder:GetChildren()) do
                    local eid = child:GetAttribute("EntityID")
                    if eid then
                        local pp = child.PrimaryPart or child:FindFirstChildWhichIsA("BasePart")
                        if pp and (pp.Position - rootPart.Position).Magnitude <= 25 then

                        end
                    end
                end
            end
        end
        if #targetList > 0 then
            swingtool(targetList)
            lastDeployableTime = ct
        end
    end)
end

local function getBushesInRange(range, fruitname)
    local bushes = {}
    if not root then return bushes end
    local charPos = root.Position
    for _, model in ipairs(workspace:GetChildren()) do
        if model:IsA("Model") and string.find(model.Name, fruitname) and model:GetAttribute("EntityID") then
            local pp = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart")
            if pp and (pp.Position - charPos).Magnitude <= range then
                table.insert(bushes, model:GetAttribute("EntityID"))
            end
        end
    end
    return bushes
end

task.spawn(function()
    while true do
        if Options.harvesttoggle.Value then
            local range = tonumber(Options.harvestrange.Value) or 30
            local fruit = Options.fruitdropdown.Value
            local bushes = getBushesInRange(range, fruit)
            for _, eid in ipairs(bushes) do
                undetectedPickup(eid)
            end
        end
        task.wait(0.1)
    end
end)

-- EVENT TAB - Auto Rebirth
Tabs.Event:AddSection("Auto Rebirth")
local autoRebirthRunning = false
local rebirthConfirmed = false

local function getPlayerLevel()
    local success, data = pcall(function() return require(rs.Modules.GameUtil) end)
    if success and data.Data and data.Data.level then
        return data.Data.level
    end
    return plr:GetAttribute("Level") or 0
end

local function doRebirth()
    if packets.Rebirth and packets.Rebirth.send then
        pcall(function() packets.Rebirth.send() end)
    end
    task.wait(3)
end

local function rebirthLoop()
    while autoRebirthRunning do
        task.wait(1)
        if getPlayerLevel() >= 100 then
            doRebirth()
            task.wait(5)
        end
    end
end

Tabs.Event:AddToggle("AutoRebirthToggle", {
    Title = "Auto Rebirth (Level 100+)",
    Default = false,
    Callback = function(state)
        if state then
            if not rebirthConfirmed then
                Window:Dialog({
                    Title = "Confirm",
                    Content = "Enable Auto Rebirth?",
                    Buttons = {
                        { Title = "Yes", Callback = function() rebirthConfirmed = true; autoRebirthRunning = true; task.spawn(rebirthLoop) end },
                        { Title = "No", Callback = function() Options.AutoRebirthToggle:SetValue(false) end }
                    }
                })
            else
                autoRebirthRunning = true
                task.spawn(rebirthLoop)
            end
        else
            autoRebirthRunning = false
        end
    end
})

-- EVENT TAB - Soul Minion TP
Tabs.Event:AddSection("Soul Minion")
Tabs.Event:AddToggle("tpsoulminionstoggle", { Title = "TP to Prioritized Targets", Default = false })

local lockedTarget = nil
local lockExpiry = 0

task.spawn(function()
    while true do
        if Options.tpsoulminionstoggle and Options.tpsoulminionstoggle.Value and root then
            local now = tick()
            if now < lockExpiry then
                task.wait(0.05)
            else
                local targetAlive = false
                if lockedTarget then
                    local crittersFolder = workspace:FindFirstChild("Critters")
                    if crittersFolder then
                        for _, c in pairs(crittersFolder:GetChildren()) do
                            if c:GetAttribute("EntityID") == lockedTarget then
                                local hum = c:FindFirstChildWhichIsA("Humanoid")
                                if hum and hum.Health > 0 then
                                    targetAlive = true
                                    local pp = c.PrimaryPart or c:FindFirstChildWhichIsA("BasePart")
                                    if pp then
                                        root.CFrame = pp.CFrame + Vector3.new(0, 3, 0)
                                    end
                                    break
                                end
                            end
                        end
                    end
                    if not targetAlive then
                        lockExpiry = now + 1.5
                        lockedTarget = nil
                    end
                end
                if not lockedTarget and now >= lockExpiry then
                    local crittersFolder = workspace:FindFirstChild("Critters")
                    if crittersFolder then
                        for _, c in pairs(crittersFolder:GetChildren()) do
                            if c:IsA("Model") and c:GetAttribute("EntityID") then
                                local name = c.Name
                                if name == "Wandering Trader" or name == "The Voodoo Spider Queen" or name == "Soul Minion" or name == "Void Ant" then
                                    lockedTarget = c:GetAttribute("EntityID")
                                    break
                                end
                            end
                        end
                    end
                end
            end
        else
            lockedTarget = nil
        end
        task.wait(0.05)
    end
end)
